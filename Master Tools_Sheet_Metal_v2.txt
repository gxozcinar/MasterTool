Sub Main()
    ' Panel zaten aciksa mevcut pencereyi one getir
    If gMainForm IsNot Nothing AndAlso Not gMainForm.IsDisposed Then
        Try
            If gMainForm.WindowState = FormWindowState.Minimized Then gMainForm.WindowState = FormWindowState.Normal
            gMainForm.Activate()
            gMainForm.BringToFront()
            Return
        Catch
        End Try
    End If

    Dim oDoc As Document = ThisApplication.ActiveDocument
    If oDoc Is Nothing OrElse oDoc.DocumentType <> DocumentTypeEnum.kAssemblyDocumentObject Then
        MessageBox.Show("Bu kural sadece Montaj (.iam) dosyalarinda calisir.", "Gecersiz Dokuman", MessageBoxButtons.OK, MessageBoxIcon.Error)
        Return
    End If

    gAsmDoc = CType(oDoc, AssemblyDocument)
    If String.IsNullOrEmpty(gAsmDoc.FullFileName) Then
        MessageBox.Show("Lutfen once montaj dosyasini kaydedin!", "Kayit Uyarisi", MessageBoxButtons.OK, MessageBoxIcon.Warning)
        Return
    End If

    BuildAndShowMainPanel()
End Sub

Private Sub SafeAdjustSplitter()
    If gSplitMain Is Nothing Then Return
    Try
        Dim totalW As Integer = gSplitMain.ClientSize.Width
        If totalW <= 1 Then Return

        Dim minPanel1 As Integer = gSplitMain.Panel1MinSize
        Dim minPanel2 As Integer = gSplitMain.Panel2MinSize
        Dim maxDistance As Integer = totalW - minPanel2
        Dim target As Integer = Math.Min(GetDxfLeftPanelWidth(), maxDistance)
        If target >= minPanel1 AndAlso target <= maxDistance Then
            gSplitMain.SplitterDistance = target
        End If
    Catch
    End Try
End Sub

Private Function GetDxfLeftPanelWidth() As Integer
    If gPartsList Is Nothing Then Return 340

    Dim fileWidth As Integer = 120
    Dim materialWidth As Integer = 105
    Dim stockWidth As Integer = 135
    Try
        For Each row As DataGridViewRow In gPartsList.Rows
            If row.Cells("PartsFileName").Value IsNot Nothing Then
                Dim fileText As String = row.Cells("PartsFileName").Value.ToString()
                fileWidth = Math.Max(fileWidth, (fileText.Length * 8) + 24)
            End If
            If row.Cells("PartsMaterial").Value IsNot Nothing Then
                Dim materialText As String = row.Cells("PartsMaterial").Value.ToString()
                materialWidth = Math.Max(materialWidth, (materialText.Length * 8) + 24)
            End If
            If row.Cells("PartsStockNumber").Value IsNot Nothing Then
                Dim stockText As String = row.Cells("PartsStockNumber").Value.ToString()
                stockWidth = Math.Max(stockWidth, (stockText.Length * 8) + 24)
            End If
        Next
    Catch
    End Try

    fileWidth = Math.Min(fileWidth, 360)
    materialWidth = Math.Min(materialWidth, 220)
    stockWidth = Math.Min(stockWidth, 240)
    Return Math.Max(620, 55 + 48 + fileWidth + materialWidth + 85 + 65 + 120 + stockWidth + 36)
End Function

Private Sub CommitGridCheckBox(sender As Object, e As EventArgs)
    Dim grid As DataGridView = TryCast(sender, DataGridView)
    If grid IsNot Nothing AndAlso grid.IsCurrentCellDirty AndAlso TypeOf grid.CurrentCell Is DataGridViewCheckBoxCell Then
        grid.CommitEdit(DataGridViewDataErrorContexts.Commit)
    End If
End Sub

Private Sub FreezeGridColumnsToContent(grid As DataGridView, minimumWidth As Integer, maximumWidth As Integer)
    If grid Is Nothing Then Return
    Try
        grid.AllowUserToResizeColumns = False
        grid.AllowUserToOrderColumns = False
        grid.AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.AllCells
        For Each column As DataGridViewColumn In grid.Columns
            If Not column.Visible Then Continue For
            column.AutoSizeMode = DataGridViewAutoSizeColumnMode.AllCells
            Dim measuredWidth As Integer = column.Width
            measuredWidth = Math.Max(minimumWidth, Math.Min(maximumWidth, measuredWidth))
            column.AutoSizeMode = DataGridViewAutoSizeColumnMode.None
            column.Width = measuredWidth
            column.HeaderCell.Style.Alignment = DataGridViewContentAlignment.MiddleCenter
            column.HeaderCell.Style.WrapMode = DataGridViewTriState.False
        Next
        grid.ColumnHeadersDefaultCellStyle.Alignment = DataGridViewContentAlignment.MiddleCenter
        grid.AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.None
        grid.ScrollBars = System.Windows.Forms.ScrollBars.Both
    Catch
    End Try
End Sub

Private Sub CalculateProfileCutting(sender As Object, e As EventArgs)
    Dim profileLength As Double
    Dim kerf As Double
    If Not Double.TryParse(gCalcProfileLength.Text.Trim().Replace(",", "."), Globalization.NumberStyles.Any, Globalization.CultureInfo.InvariantCulture, profileLength) OrElse profileLength <= 0 Then
        MessageBox.Show("Profil uzunluğu 0'dan büyük olmalıdır.", "Hesaplama", MessageBoxButtons.OK, MessageBoxIcon.Warning)
        Return
    End If
    If Not Double.TryParse(gCalcKerf.Text.Trim().Replace(",", "."), Globalization.NumberStyles.Any, Globalization.CultureInfo.InvariantCulture, kerf) OrElse kerf < 0 Then
        MessageBox.Show("Kerf değeri 0 veya daha büyük olmalıdır.", "Hesaplama", MessageBoxButtons.OK, MessageBoxIcon.Warning)
        Return
    End If

    Dim pieces As New List(Of CalcPiece)
    Dim requestedPieceCount As Integer = 0
    For Each row As DataGridViewRow In gCalcGrid.Rows
        Dim length As Double
        Dim qty As Integer
        If String.IsNullOrWhiteSpace(ValueToText(row.Cells("CalcName").Value)) AndAlso String.IsNullOrWhiteSpace(ValueToText(row.Cells("CalcCode").Value)) Then Continue For
        If Not Double.TryParse(ValueToText(row.Cells("CalcLength").Value).Replace(",", "."), Globalization.NumberStyles.Any, Globalization.CultureInfo.InvariantCulture, length) OrElse length <= 0 Then
            MessageBox.Show("Geçersiz parça uzunluğu: satır " & (row.Index + 1).ToString(), "Hesaplama", MessageBoxButtons.OK, MessageBoxIcon.Warning)
            Return
        End If
        If Not Integer.TryParse(ValueToText(row.Cells("CalcQty").Value), qty) OrElse qty < 1 Then
            MessageBox.Show("Parça adedi 1 veya daha büyük tam sayı olmalıdır: satır " & (row.Index + 1).ToString(), "Hesaplama", MessageBoxButtons.OK, MessageBoxIcon.Warning)
            Return
        End If
        If length + kerf > profileLength Then
            MessageBox.Show("HATA: " & ValueToText(row.Cells("CalcCode").Value) & " kodlu parça, kesim payı ile profile sığmıyor.", "Hesaplama", MessageBoxButtons.OK, MessageBoxIcon.Error)
            Return
        End If
        requestedPieceCount += qty
        For i As Integer = 1 To qty
            pieces.Add(New CalcPiece(ValueToText(row.Cells("CalcName").Value), ValueToText(row.Cells("CalcCode").Value), length))
        Next
    Next
    If pieces.Count = 0 Then
        MessageBox.Show("En az bir parça ekleyin.", "Hesaplama", MessageBoxButtons.OK, MessageBoxIcon.Warning)
        Return
    End If

    pieces.Sort(Function(a, b) b.Length.CompareTo(a.Length))
    Dim bars As New List(Of CalcBar)
    For Each piece As CalcPiece In pieces
        Dim best As CalcBar = Nothing
        Dim bestRemaining As Double = Double.MaxValue
        For Each bar As CalcBar In bars
            Dim required As Double = piece.Length + kerf
            If required <= bar.Remaining AndAlso bar.Remaining - required < bestRemaining Then
                best = bar
                bestRemaining = bar.Remaining - required
            End If
        Next
        If best Is Nothing Then
            best = New CalcBar(bars.Count + 1, profileLength)
            bars.Add(best)
        End If
        best.Pieces.Add(piece)
        best.Remaining -= piece.Length + kerf
    Next

    Dim partTotal As Double = 0
    For Each p As CalcPiece In pieces : partTotal += p.Length : Next
    Dim totalKerf As Double = pieces.Count * kerf
    Dim totalStock As Double = bars.Count * profileLength
    Dim waste As Double = totalStock - partTotal - totalKerf
    Dim efficiency As Double = If(totalStock <= 0, 0, (partTotal + totalKerf) / totalStock * 100)
    gCalcSummary.Text = "Profil: " & bars.Count & " adet | Stok: " & totalStock.ToString("0.##") & " mm | Parça: " & partTotal.ToString("0.##") & " mm | Kerf: " & totalKerf.ToString("0.##") & " mm | Fire: " & waste.ToString("0.##") & " mm | Verim: %" & efficiency.ToString("0.##")

    Dim result As New StringBuilder()
    result.AppendLine("PROFİL KESİM PLANI - " & gCalcProfileName.Text)
    result.AppendLine("============================================================")
    result.AppendLine("TOPLAM GEREKLİ PROFİL ADEDİ : " & bars.Count.ToString())
    result.AppendLine("TOPLAM STOK UZUNLUĞU         : " & totalStock.ToString("0.##") & " mm")
    result.AppendLine("TOPLAM PARÇA UZUNLUĞU        : " & partTotal.ToString("0.##") & " mm")
    result.AppendLine("TOPLAM KESİM PAYI (KERF)     : " & totalKerf.ToString("0.##") & " mm")
    result.AppendLine("TOPLAM FIRE                  : " & waste.ToString("0.##") & " mm")
    result.AppendLine("VERİMLİLİK                   : %" & efficiency.ToString("0.##"))
    result.AppendLine("FORMÜL: Profil başına kullanım = Parça toplamı + (Kesim adedi × Kerf)")
    result.AppendLine("FORMÜL: Toplam fire = (Profil adedi × Profil boyu) - Parça toplamı - Toplam kerf")
    result.AppendLine("Profil boyu: " & profileLength.ToString("0.##") & " mm | Kerf: " & kerf.ToString("0.##") & " mm")
    result.AppendLine()
    For Each bar As CalcBar In bars
        Dim barPartLength As Double = 0
        For Each barPiece As CalcPiece In bar.Pieces
            barPartLength += barPiece.Length
        Next
        Dim barKerf As Double = bar.Pieces.Count * kerf
        result.AppendLine("------------------------------------------------------------")
        result.AppendLine("PROFİL #" & bar.Index.ToString() & " | KESİLECEK PARÇALAR")
        result.AppendLine("Parça toplamı: " & barPartLength.ToString("0.##") & " mm | Kesim adedi: " & bar.Pieces.Count.ToString() & " | Kerf: " & barKerf.ToString("0.##") & " mm")
        result.AppendLine("Kullanılan: " & (barPartLength + barKerf).ToString("0.##") & " mm | Kalan fire: " & bar.Remaining.ToString("0.##") & " mm")
        Dim order As Integer = 0
        For Each piece As CalcPiece In bar.Pieces
            order += 1
            result.AppendLine("  " & order.ToString() & ". " & piece.Length.ToString("0.##") & " mm | " & piece.Name & " | " & piece.Code)
        Next
        result.AppendLine()
    Next
    result.AppendLine("------------------------------------------------------------")
    result.AppendLine("KONTROL: İstenen parça adedi = " & requestedPieceCount.ToString() & " | Yerleştirilen parça adedi = " & pieces.Count.ToString() & " | Durum: " & If(requestedPieceCount = pieces.Count, "TAM", "HATALI"))
    gCalcResult.Text = result.ToString()
End Sub

Private Class CalcPiece
    Public Name As String
    Public Code As String
    Public Length As Double
    Public Sub New(n As String, c As String, l As Double)
        Name = n : Code = c : Length = l
    End Sub
End Class

Private Class CalcBar
    Public Index As Integer
    Public Remaining As Double
    Public Pieces As New List(Of CalcPiece)
    Public Sub New(i As Integer, length As Double)
        Index = i : Remaining = length
    End Sub
End Class

Private Sub UpdatePartSummary()
    If gLblPartCount Is Nothing OrElse gMatGrid Is Nothing Then Return
    Dim total As Integer = 0, visible As Integer = 0, selected As Integer = 0
    For Each row As DataGridViewRow In gMatGrid.Rows
        total += 1
        If row.Visible Then visible += 1
        Try
            If Convert.ToBoolean(row.Cells("MaterialExportSelected").Value) Then selected += 1
        Catch
        End Try
    Next
    gLblPartCount.Text = "Toplam: " & total & " | Görünen: " & visible & " | Seçili: " & selected
End Sub

Private Function GetRowProductionTypeCode(row As DataGridViewRow) As String
    If row Is Nothing Then Return ""
    Return NormalizeProductionType(ValueToText(row.Cells("ProductionType").Value))
End Function

Private Sub UpdateRowStatus(rowIndex As Integer)
    If gMatGrid Is Nothing OrElse rowIndex < 0 OrElse rowIndex >= gMatGrid.Rows.Count Then Return
    Dim row As DataGridViewRow = gMatGrid.Rows(rowIndex)
    Dim filePath As String = ValueToText(row.Cells("FilePath").Value)
    Dim pDoc As PartDocument = Nothing
    If gUniqueParts Is Nothing OrElse Not gUniqueParts.TryGetValue(filePath, pDoc) Then Return

    Dim thicknessText As String = ValueToText(row.Cells("Thickness").Value)
    Dim productionType As String = GetRowProductionTypeCode(row)
    Dim enteredThickness As Double = 0
    If TryParseThickness(thicknessText, enteredThickness) Then
        productionType = PT_SHEETMETAL
        Dim displayType As String = ProductionTypeToDisplay(PT_SHEETMETAL)
        If ValueToText(row.Cells("ProductionType").Value) <> displayType Then row.Cells("ProductionType").Value = displayType
    End If

    row.Cells("Status").Value = GetPartStatus(pDoc, productionType, thicknessText)
End Sub

Private Sub TransferSelectedPartsToDxfTab()
    If gMatGrid Is Nothing OrElse Not gMatGrid.EndEdit() Then Return
    gSelectedForDxf.Clear()
    Dim skipped As Integer = 0
    For Each row As DataGridViewRow In gMatGrid.Rows
        Dim selected As Boolean = False
        Try : selected = Convert.ToBoolean(row.Cells("MaterialExportSelected").Value) : Catch : End Try
        If Not selected Then Continue For
        Dim filePath As String = ValueToText(row.Cells("FilePath").Value)
        Dim pDoc As PartDocument = Nothing
        If gUniqueParts IsNot Nothing AndAlso gUniqueParts.TryGetValue(filePath, pDoc) Then
            If GetRowProductionTypeCode(row) = PT_SHEETMETAL AndAlso IsSheetMetalPart(pDoc) Then
                gSelectedForDxf.Add(filePath)
            Else
                skipped += 1
            End If
        End If
    Next
    If gSelectedForDxf.Count = 0 Then
        MessageBox.Show("DXF/IDW sayfasına aktarılabilecek hazır Sac Metal parça yok." & vbCrLf & _
                        "Normal/STEP parçalara kalınlık girip 'Değişiklikleri Uygula' işlemini çalıştırın.", _

Private Sub OpenPartFileLocation(filePath As String)
    If String.IsNullOrWhiteSpace(filePath) Then Return
    If Not System.IO.File.Exists(filePath) Then
        MessageBox.Show("Dosya bulunamadı:" & vbCrLf & filePath, "Dosya Konumu", MessageBoxButtons.OK, MessageBoxIcon.Warning)
        Return
    End If
    Try
        Dim psi As New System.Diagnostics.ProcessStartInfo()
        psi.FileName = "explorer.exe"
        psi.Arguments = "/select," & Chr(34) & filePath & Chr(34)
        psi.UseShellExecute = True
        System.Diagnostics.Process.Start(psi)
    Catch ex As Exception
        MessageBox.Show("Dosya konumu açılamadı: " & ex.Message, "Dosya Konumu", MessageBoxButtons.OK, MessageBoxIcon.Warning)
    End Try
End Sub

Private Sub OpenPartDrawing(filePath As String)
    If String.IsNullOrEmpty(filePath) Then Return
    Dim directPath As String = System.IO.Path.ChangeExtension(filePath, ".idw")
    Dim found As String = ""
    If System.IO.File.Exists(directPath) Then found = directPath
    If found = "" Then
        Try
            Dim outputFolder As String = GetSharedOutputFolder()
            Dim partBase As String = SanitizeFileName(System.IO.Path.GetFileNameWithoutExtension(filePath))
            Dim candidates As String() = System.IO.Directory.GetFiles(outputFolder, "*" & partBase & "*.idw", System.IO.SearchOption.AllDirectories)
            If candidates.Length > 0 Then found = candidates(0)
        Catch
        End Try
    End If
    If found = "" Then
        MessageBox.Show("Bu parça için mevcut IDW bulunamadı.", "Teknik Resim", MessageBoxButtons.OK, MessageBoxIcon.Information)
        Return
    End If
    Try : ThisApplication.Documents.Open(found, True) : Catch ex As Exception : MessageBox.Show(ex.Message) : End Try
End Sub

Private Sub HandleMaterialClipboardKey(e As KeyEventArgs)
    If e Is Nothing OrElse e.Handled OrElse Not e.Control Then Return
    If e.KeyCode <> Keys.C AndAlso e.KeyCode <> Keys.V Then Return
    If gMatGrid Is Nothing OrElse gMatGrid.CurrentCell Is Nothing Then Return
    Try
        Dim editBox As System.Windows.Forms.TextBox = TryCast(gMatGrid.EditingControl, System.Windows.Forms.TextBox)
        If editBox IsNot Nothing AndAlso editBox.Focused Then
            If e.KeyCode = Keys.C Then
                If editBox.SelectionLength > 0 Then
                    editBox.Copy()
                Else
                    Clipboard.SetText(editBox.Text)
                End If
            ElseIf Not gMatGrid.CurrentCell.ReadOnly Then
                Dim pasteText As String = Clipboard.GetText()
                If pasteText.Contains(vbTab) OrElse pasteText.Contains(vbCr) OrElse pasteText.Contains(vbLf) Then
                    If gMatGrid.EndEdit() Then PasteGridClipboard(gMatGrid)
                Else
                    editBox.Paste()
                End If
            End If
        Else
            GridClipboard_KeyDown(gMatGrid, e)
            Return
        End If
        e.SuppressKeyPress = True
        e.Handled = True
    Catch ex As Exception
        AppendLog("[UYARI] Kopyala/yapıştır işlemi başarısız: " & ex.Message)
    End Try
End Sub

Private Sub GridClipboard_KeyDown(sender As Object, e As KeyEventArgs)
    Dim grid As DataGridView = TryCast(sender, DataGridView)
    If grid Is Nothing OrElse grid.CurrentCell Is Nothing OrElse e.Handled OrElse Not e.Control Then Return
    Try
        If e.KeyCode = Keys.C Then
            If grid.GetCellCount(DataGridViewElementStates.Selected) > 0 Then
                Clipboard.SetDataObject(grid.GetClipboardContent())
            End If
        ElseIf e.KeyCode = Keys.V Then
            PasteGridClipboard(grid)
        ElseIf e.KeyCode = Keys.X Then
            If grid.GetCellCount(DataGridViewElementStates.Selected) > 0 Then
                Clipboard.SetDataObject(grid.GetClipboardContent())
                ClearEditableSelectedCells(grid)
            End If
        ElseIf e.KeyCode = Keys.Z Then
            UndoGridCellEdit()
        Else
            Return
        End If
        e.SuppressKeyPress = True
        e.Handled = True
    Catch ex As Exception
        AppendLog("[UYARI] Kopyala/yapıştır işlemi başarısız: " & ex.Message)
    End Try
End Sub

Private Sub PasteGridClipboard(grid As DataGridView)
    Dim text As String = Clipboard.GetText()
    If String.IsNullOrWhiteSpace(text) OrElse grid.CurrentCell Is Nothing Then Return
    Dim rows() As String = text.Replace(vbCrLf, vbLf).Replace(vbCr, vbLf).Split(ControlChars.Lf)
    Dim startRow As Integer = grid.CurrentCell.RowIndex
    Dim startDisplayIndex As Integer = grid.CurrentCell.OwningColumn.DisplayIndex
    For r As Integer = 0 To rows.Length - 1
        If startRow + r >= grid.Rows.Count OrElse rows(r) = "" Then Continue For
        Dim values() As String = rows(r).Split(New Char() {ControlChars.Tab})
        Dim nextDisplayIndex As Integer = startDisplayIndex
        For c As Integer = 0 To values.Length - 1
            Dim targetColumn As DataGridViewColumn = Nothing
            While nextDisplayIndex < grid.Columns.Count AndAlso targetColumn Is Nothing
                For Each candidate As DataGridViewColumn In grid.Columns
                    If candidate.Visible AndAlso candidate.DisplayIndex = nextDisplayIndex Then
                        targetColumn = candidate
                        Exit For
                    End If
                Next
                nextDisplayIndex += 1
            End While
            If targetColumn Is Nothing Then Exit For
            Dim cell As DataGridViewCell = grid.Rows(startRow + r).Cells(targetColumn.Index)
            If Not cell.OwningColumn.ReadOnly AndAlso TypeOf cell Is DataGridViewTextBoxCell Then cell.Value = values(c).Trim()
        Next
    Next
    If grid Is gMatGrid Then ApplyRowColor(startRow)
End Sub

Private Sub ClearEditableSelectedCells(grid As DataGridView)
    For Each cell As DataGridViewCell In grid.SelectedCells
        If Not cell.OwningColumn.ReadOnly AndAlso TypeOf cell Is DataGridViewTextBoxCell Then cell.Value = String.Empty
    Next
End Sub

End Sub

Private Sub OpenIPropertiesForPart(filePath As String, rowIndex As Integer)
    Dim wasTopMost As Boolean = gMainForm.TopMost
    gMainForm.TopMost = False
    Dim inPlaceEditSuccess As Boolean = False
    Dim oOcc As ComponentOccurrence = Nothing

    Try
        If gPartOccurrences IsNot Nothing Then gPartOccurrences.TryGetValue(filePath, oOcc)

        If oOcc IsNot Nothing Then
            Try
                gAsmDoc.Activate()
                System.Windows.Forms.Application.DoEvents()
                oOcc.Edit()
                System.Windows.Forms.Application.DoEvents()
                inPlaceEditSuccess = True
            Catch ex As Exception
                inPlaceEditSuccess = False
            End Try
        End If

        If Not inPlaceEditSuccess Then
            Try
                Dim pDocTemp As Document = ThisApplication.Documents.Open(filePath, True)
                pDocTemp.Activate()
                System.Windows.Forms.Application.DoEvents()
            Catch ex As Exception
                MessageBox.Show("Parça açılamadı: " & ex.Message, "Uyarı", MessageBoxButtons.OK, MessageBoxIcon.Exclamation)
                Return
            End Try
        End If

        ThisApplication.UserInterfaceManager.DoEvents()
        System.Threading.Thread.Sleep(100)
        System.Windows.Forms.Application.DoEvents()

        Dim oCtrlDef As ControlDefinition = Nothing
        Dim cmdNames As String() = {"AppiPropertiesWrapperCmd", "PartiPropertiesCmd", "AppFilePropertiesCmd"}
        For Each cName As String In cmdNames
            Try
                oCtrlDef = ThisApplication.CommandManager.ControlDefinitions.Item(cName)
                If oCtrlDef IsNot Nothing Then Exit For
            Catch
            End Try
        Next

        If oCtrlDef IsNot Nothing Then
            Try
                oCtrlDef.Execute2(True)
            Catch
                Try : oCtrlDef.Execute() : Catch : End Try
            End Try
        Else
            MessageBox.Show("iProperties komutu bulunamadı.", "Uyarı", MessageBoxButtons.OK, MessageBoxIcon.Exclamation)
        End If

        If inPlaceEditSuccess AndAlso oOcc IsNot Nothing Then
            Try
                oOcc.ExitEdit(ExitTypeEnum.kExitToTop)
            Catch
                Try
                    Dim topDef As AssemblyComponentDefinition = gAsmDoc.ComponentDefinition
                    If topDef.Occurrences.Count > 0 Then topDef.Occurrences(1).ExitEdit(ExitTypeEnum.kExitToTop)
                Catch
                End Try
            End Try
        End If

        Try : gAsmDoc.Activate() : Catch : End Try

        RefreshGridRowForPart(rowIndex, filePath)

    Catch ex As Exception
        MessageBox.Show("iProperties işlemi sırasında hata oluştu: " & ex.Message, "Hata", MessageBoxButtons.OK, MessageBoxIcon.Error)
    Finally
        gMainForm.TopMost = wasTopMost
        gMainForm.Activate()
    End Try
End Sub


Private Sub RefreshGridRowForPart(rowIndex As Integer, filePath As String)
    If gMatGrid Is Nothing OrElse rowIndex < 0 OrElse rowIndex >= gMatGrid.Rows.Count Then Return
    ReadBomQuantities()
    Dim refreshedPart As PartDocument = Nothing
    If gUniqueParts IsNot Nothing AndAlso gUniqueParts.TryGetValue(filePath, refreshedPart) AndAlso IsExcludedBomStructure(refreshedPart) Then
        PopulateMaterialGrid()
        Return
    End If
    Dim pendingBefore As Boolean = gBomGridHasPendingEdits
    gPopulatingBomGrid = True
    Try
        Dim pDoc As PartDocument = Nothing
        If gUniqueParts IsNot Nothing AndAlso gUniqueParts.TryGetValue(filePath, pDoc) Then
            Dim description As String = GetDesignTrackingProperty(pDoc, "Description", "")
            Dim currentMaterial As String = ""
            Try : currentMaterial = pDoc.ActiveMaterial.DisplayName : Catch : End Try
            Dim currentThickness As String = GetPartThicknessText(pDoc).Replace(" mm", "")
            Dim prodType As String = GetProductionType(pDoc)

            Dim row As DataGridViewRow = gMatGrid.Rows(rowIndex)
            row.Cells("FileName").Value = System.IO.Path.GetFileName(filePath)
            Dim previousImage As System.IDisposable = TryCast(row.Cells("Thumbnail").Value, System.IDisposable)
            row.Cells("Thumbnail").Value = GetPartThumbnail(pDoc)
            If previousImage IsNot Nothing Then previousImage.Dispose()
            row.Cells("PartName").Value = GetDesignTrackingProperty(pDoc, "Part Number", pDoc.DisplayName)
            row.Cells("Description").Value = description
            row.Cells("StockNumber").Value = GetDesignTrackingProperty(pDoc, "Stock Number", "")
            Dim bomQuantity As String = Nothing
            If gBomQuantities IsNot Nothing AndAlso gBomQuantities.TryGetValue(filePath, bomQuantity) Then
                row.Cells("ItemQuantity").Value = bomQuantity
            Else
                row.Cells("ItemQuantity").Value = GetPartOccurrenceCount(filePath)
            End If
            SetMaterialChoices(row, pDoc, currentMaterial)
            row.Cells("BOMStructure").Value = BomStructureToDisplay(GetEffectiveBomStructure(pDoc))
            row.Cells("Thickness").Value = currentThickness
            row.Cells("ProductionType").Value = ProductionTypeToDisplay(prodType)
            row.Cells("Status").Value = GetPartStatus(pDoc, prodType, currentThickness)
            CaptureBomRowValues(row)
            ApplyRowColor(rowIndex)
            UpdatePartSummary()
        End If
    Catch
    Finally
        gPopulatingBomGrid = False
        gBomGridHasPendingEdits = pendingBefore
    End Try
End Sub

Private Function GetDesignTrackingProperty(oDoc As Document, propertyName As String, defaultValue As String) As String
    If oDoc Is Nothing Then Return defaultValue
    Try
        Dim value As Object = oDoc.PropertySets("Design Tracking Properties")(propertyName).Value
        If value IsNot Nothing AndAlso Not String.IsNullOrEmpty(value.ToString().Trim()) Then
            Return value.ToString()
        End If
    Catch
    End Try
    Return defaultValue
End Function

Private Sub SetDesignTrackingPropertyIfChanged(oDoc As Document, propertyName As String, value As String)
    Dim currentValue As String = GetDesignTrackingProperty(oDoc, propertyName, "")
    If String.Equals(currentValue, value, StringComparison.Ordinal) Then Return
    oDoc.PropertySets("Design Tracking Properties")(propertyName).Value = value
End Sub


Private Sub PopulateMaterialGrid()
    If gMatGrid Is Nothing Then Return
    If gMatGrid.IsCurrentCellInEditMode AndAlso Not gMatGrid.EndEdit() Then Return
    System.Windows.Forms.Cursor.Current = System.Windows.Forms.Cursors.WaitCursor
    Dim layoutSuspended As Boolean = False
    Try

    Dim hadExistingRows As Boolean = gMatGrid.Rows.Count > 0
    Dim selectedBefore As New HashSet(Of String)(StringComparer.OrdinalIgnoreCase)
    Dim formatBefore As New Dictionary(Of String, String)(StringComparer.OrdinalIgnoreCase)
    For Each oldRow As DataGridViewRow In gMatGrid.Rows
        Try
            Dim oldPath As String = ValueToText(oldRow.Cells("FilePath").Value)
            If Convert.ToBoolean(oldRow.Cells("MaterialExportSelected").Value) Then
                selectedBefore.Add(oldPath)
            End If
            formatBefore(oldPath) = ValueToText(oldRow.Cells("MaterialExportFormat").Value)
        Catch
        End Try
    Next

    gUniqueParts = New Dictionary(Of String, PartDocument)(StringComparer.OrdinalIgnoreCase)
    gPartOccurrences = New Dictionary(Of String, ComponentOccurrence)(StringComparer.OrdinalIgnoreCase)
    gPartCounts = New Dictionary(Of String, Integer)(StringComparer.OrdinalIgnoreCase)
    Dim oAssyDef As AssemblyComponentDefinition = gAsmDoc.ComponentDefinition
    ReadBomQuantities()
    gMaterialLibraryChoices = New List(Of String)()
    Try
        If gMatLib IsNot Nothing Then
            For Each asset As MaterialAsset In gMatLib.MaterialAssets
                If Not String.IsNullOrWhiteSpace(asset.DisplayName) Then gMaterialLibraryChoices.Add(asset.DisplayName)
            Next
        End If
    Catch
    End Try

    ' Listeye yalnızca Parts Only BOM içindeki Normal satırları alınır.
    For Each kvp As KeyValuePair(Of String, PartDocument) In gBomParts
        gUniqueParts.Add(kvp.Key, kvp.Value)
        Dim bomQty As String = Nothing
        Dim count As Integer = 1
        If gBomQuantities.TryGetValue(kvp.Key, bomQty) Then Integer.TryParse(bomQty, count)
        gPartCounts(kvp.Key) = Math.Max(1, count)
    Next

    ' Occurrence yalnızca yerinde düzenleme için kullanılır; listeye parça eklemez.
    For Each oOcc As ComponentOccurrence In oAssyDef.Occurrences.AllLeafOccurrences
        Try
            If oOcc.Suppressed OrElse oOcc.DefinitionDocumentType <> DocumentTypeEnum.kPartDocumentObject Then Continue For
            Dim oPartDoc As PartDocument = TryCast(oOcc.Definition.Document, PartDocument)
            If oPartDoc IsNot Nothing AndAlso gUniqueParts.ContainsKey(oPartDoc.FullFileName) AndAlso Not gPartOccurrences.ContainsKey(oPartDoc.FullFileName) Then
                gPartOccurrences.Add(oPartDoc.FullFileName, oOcc)
            End If
        Catch
        End Try
    Next

    gPopulatingBomGrid = True
    gMatGrid.SuspendLayout()
    layoutSuspended = True
    For Each oldRow As DataGridViewRow In gMatGrid.Rows
        Dim oldImage As System.IDisposable = TryCast(oldRow.Cells("Thumbnail").Value, System.IDisposable)
        If oldImage IsNot Nothing Then oldImage.Dispose()
    Next
    gMatGrid.Rows.Clear()
    Dim rowNo As Integer = 0

    For Each kvp As KeyValuePair(Of String, PartDocument) In gUniqueParts
        Dim filePath As String = kvp.Key
        Dim pDoc As PartDocument = kvp.Value
        If IsExcludedBomStructure(pDoc) Then Continue For
        rowNo += 1
        Dim description As String = GetDesignTrackingProperty(pDoc, "Description", "")
        Dim currentMaterial As String = ""
        Try : currentMaterial = pDoc.ActiveMaterial.DisplayName : Catch : End Try
        Dim currentThickness As String = GetPartThicknessText(pDoc).Replace(" mm", "")
        Dim stockNumber As String = GetDesignTrackingProperty(pDoc, "Stock Number", "")
        Dim partNumber As String = GetDesignTrackingProperty(pDoc, "Part Number", pDoc.DisplayName)
        Dim prodType As String = GetProductionType(pDoc)

        Dim idx As Integer = gMatGrid.Rows.Add()
        Dim row As DataGridViewRow = gMatGrid.Rows(idx)
        row.Tag = filePath
        row.Cells("No").Value = rowNo
        row.Cells("FilePath").Value = filePath
        row.Cells("FileName").Value = System.IO.Path.GetFileName(filePath)
        row.Cells("Thumbnail").Value = GetPartThumbnail(pDoc)
        row.Cells("PartName").Value = partNumber
        row.Cells("Description").Value = description
        row.Cells("StockNumber").Value = stockNumber
        Dim bomQuantity As String = Nothing
        If gBomQuantities IsNot Nothing AndAlso gBomQuantities.TryGetValue(filePath, bomQuantity) Then
            row.Cells("ItemQuantity").Value = bomQuantity
        Else
            row.Cells("ItemQuantity").Value = GetPartOccurrenceCount(filePath)
        End If
        SetMaterialChoices(row, pDoc, currentMaterial)
        row.Cells("BOMStructure").Value = BomStructureToDisplay(GetEffectiveBomStructure(pDoc))
        row.Cells("Thickness").Value = currentThickness
        row.Cells("ProductionType").Value = ProductionTypeToDisplay(prodType)
        row.Cells("Status").Value = GetPartStatus(pDoc, prodType, currentThickness)
        Dim savedFormat As String = Nothing
        If formatBefore.TryGetValue(filePath, savedFormat) AndAlso System.Array.IndexOf(gExportFormatItems, savedFormat) >= 0 Then
            row.Cells("MaterialExportFormat").Value = savedFormat
        Else
            row.Cells("MaterialExportFormat").Value = gExportFormatItems(0)
        End If
        row.Cells("MaterialExportSelected").Value = If(hadExistingRows, selectedBefore.Contains(filePath), True)
        CaptureBomRowValues(row)
        ApplyRowColor(idx)
    Next

    LayoutBomGridColumns()
    gBomGridHasPendingEdits = False
    UpdatePartSummary()
    Catch ex As Exception
        AppendLog("[HATA] Parça listesi oluşturulamadı: " & ex.Message)
        MessageBox.Show(ex.Message, "Liste Yenileme", MessageBoxButtons.OK, MessageBoxIcon.Error)
    Finally
        If layoutSuspended Then gMatGrid.ResumeLayout()
        gPopulatingBomGrid = False
        System.Windows.Forms.Cursor.Current = System.Windows.Forms.Cursors.Default
    End Try
End Sub

Private Function IsExcludedBomStructure(pDoc As PartDocument) As Boolean
    If pDoc Is Nothing OrElse gBomStructures Is Nothing Then Return True
    Dim bomKind As BOMStructureEnum
    Return Not gBomStructures.TryGetValue(pDoc.FullFileName, bomKind) OrElse bomKind <> BOMStructureEnum.kNormalBOMStructure
End Function

Private Function GetEffectiveBomStructure(pDoc As PartDocument) As BOMStructureEnum
    If pDoc Is Nothing Then Return BOMStructureEnum.kNormalBOMStructure
    Dim bomKind As BOMStructureEnum
    If gBomStructures IsNot Nothing AndAlso gBomStructures.TryGetValue(pDoc.FullFileName, bomKind) Then
        If bomKind <> BOMStructureEnum.kDefaultBOMStructure AndAlso bomKind <> BOMStructureEnum.kVariesBOMStructure Then Return bomKind
    End If
    Try
        Dim occurrence As ComponentOccurrence = Nothing
        If gPartOccurrences IsNot Nothing AndAlso gPartOccurrences.TryGetValue(pDoc.FullFileName, occurrence) Then
            bomKind = occurrence.BOMStructure
            If bomKind <> BOMStructureEnum.kDefaultBOMStructure Then Return bomKind
        End If
    Catch
    End Try
    Try : Return pDoc.ComponentDefinition.BOMStructure : Catch : Return BOMStructureEnum.kNormalBOMStructure : End Try
End Function

Private Sub SetMaterialChoices(row As DataGridViewRow, pDoc As PartDocument, currentMaterial As String)
    Dim materialCell As DataGridViewComboBoxCell = TryCast(row.Cells("Material"), DataGridViewComboBoxCell)
    If materialCell Is Nothing Then Return
    Dim names As New HashSet(Of String)(StringComparer.OrdinalIgnoreCase)
    If Not String.IsNullOrWhiteSpace(currentMaterial) Then names.Add(currentMaterial)
    Try
        For Each asset As MaterialAsset In pDoc.MaterialAssets
            If Not String.IsNullOrWhiteSpace(asset.DisplayName) Then names.Add(asset.DisplayName)
        Next
    Catch
    End Try
    If gMaterialLibraryChoices IsNot Nothing Then
        For Each materialName As String In gMaterialLibraryChoices
            names.Add(materialName)
        Next
    End If
    Dim choices As New List(Of String)(names)
    choices.Sort(StringComparer.CurrentCultureIgnoreCase)
    materialCell.Items.Clear()
    materialCell.Items.AddRange(choices.ToArray())
    materialCell.Value = currentMaterial
End Sub

Private Sub CaptureBomRowValues(row As DataGridViewRow)
    For Each fieldName As String In New String() {"PartName", "Description", "StockNumber", "Material", "BOMStructure", "Thickness", "ProductionType"}

Private Function BomCellWasEdited(row As DataGridViewRow, fieldName As String) As Boolean

Private Function RowHasPendingChanges(row As DataGridViewRow) As Boolean

Private Sub SyncBomGridIfChanged(sender As Object, e As EventArgs)
    If gMatGrid Is Nothing OrElse gAsmDoc Is Nothing OrElse gPopulatingBomGrid OrElse gBomGridHasPendingEdits OrElse gBomSyncInProgress Then Return
    If gMatGrid.IsCurrentCellInEditMode OrElse gMainForm Is Nothing OrElse gMainForm.IsDisposed Then Return
    gBomSyncInProgress = True
    Try
        ReadBomQuantities()
        If gUniqueParts Is Nothing OrElse gBomParts.Count <> gUniqueParts.Count Then
            PopulateMaterialGrid()
            Exit Sub
        End If
        For Each filePath As String In gBomParts.Keys
            If Not gUniqueParts.ContainsKey(filePath) Then
                PopulateMaterialGrid()
                Exit Sub
            End If
        Next
        Dim expectedRows As Integer = 0
        If gUniqueParts IsNot Nothing Then
            For Each knownPart As PartDocument In gUniqueParts.Values
                If Not IsExcludedBomStructure(knownPart) Then expectedRows += 1
            Next
        End If
        If expectedRows <> gMatGrid.Rows.Count Then
            PopulateMaterialGrid()
            Exit Sub
        End If
        For Each row As DataGridViewRow In gMatGrid.Rows
            Dim filePath As String = ValueToText(row.Cells("FilePath").Value)
            Dim pDoc As PartDocument = Nothing
            If gUniqueParts Is Nothing OrElse Not gUniqueParts.TryGetValue(filePath, pDoc) Then Continue For
            Dim materialName As String = ""
            Try : materialName = pDoc.ActiveMaterial.DisplayName : Catch : End Try
            Dim quantity As String = Nothing
            If gBomQuantities Is Nothing OrElse Not gBomQuantities.TryGetValue(filePath, quantity) Then quantity = GetPartOccurrenceCount(filePath).ToString()
            If ValueToText(row.Cells("PartName").Value) <> GetDesignTrackingProperty(pDoc, "Part Number", pDoc.DisplayName) OrElse _
               ValueToText(row.Cells("Description").Value) <> GetDesignTrackingProperty(pDoc, "Description", "") OrElse _
               ValueToText(row.Cells("StockNumber").Value) <> GetDesignTrackingProperty(pDoc, "Stock Number", "") OrElse _
               ValueToText(row.Cells("Material").Value) <> materialName OrElse _
               ValueToText(row.Cells("BOMStructure").Value) <> BomStructureToDisplay(GetEffectiveBomStructure(pDoc)) OrElse _
               ValueToText(row.Cells("ItemQuantity").Value) <> quantity OrElse _
               ValueToText(row.Cells("Thickness").Value) <> GetPartThicknessText(pDoc).Replace(" mm", "") OrElse _
               ValueToText(row.Cells("ProductionType").Value) <> ProductionTypeToDisplay(GetProductionType(pDoc)) Then
                PopulateMaterialGrid()
                Exit Sub
            End If
        Next
    Catch ex As Exception
        AppendLog("[UYARI] BOM senkronizasyonu: " & ex.Message)
    Finally
        gBomSyncInProgress = False
    End Try
End Sub

Private Sub ReadBomQuantities()
    Dim bomParts As New Dictionary(Of String, PartDocument)(StringComparer.OrdinalIgnoreCase)
    Dim quantities As New Dictionary(Of String, String)(StringComparer.OrdinalIgnoreCase)
    Dim structures As New Dictionary(Of String, BOMStructureEnum)(StringComparer.OrdinalIgnoreCase)
    Try
        Dim bom As BOM = gAsmDoc.ComponentDefinition.BOM
        If Not bom.PartsOnlyViewEnabled Then bom.PartsOnlyViewEnabled = True
        For Each bomView As BOMView In bom.BOMViews
            If bomView.ViewType <> BOMViewTypeEnum.kPartsOnlyBOMViewType Then Continue For
            For Each bomRow As BOMRow In bomView.BOMRows
                If bomRow.BOMStructure <> BOMStructureEnum.kNormalBOMStructure Then Continue For
                For Each definition As ComponentDefinition In bomRow.ComponentDefinitions
                    Dim partDoc As PartDocument = TryCast(definition.Document, PartDocument)
                    If partDoc Is Nothing OrElse String.IsNullOrEmpty(partDoc.FullFileName) Then Continue For
                    Dim filePath As String = partDoc.FullFileName
                    If Not bomParts.ContainsKey(filePath) Then bomParts.Add(filePath, partDoc)
                    quantities(filePath) = CStr(bomRow.TotalQuantity)
                    structures(filePath) = bomRow.BOMStructure
                Next
            Next
            Exit For
        Next
        gBomParts = bomParts
        gBomQuantities = quantities
        gBomStructures = structures
    Catch ex As Exception
        AppendLog("[UYARI] Parts Only BOM okunamadı: " & ex.Message)
        If gBomParts Is Nothing Then gBomParts = New Dictionary(Of String, PartDocument)(StringComparer.OrdinalIgnoreCase)
        If gBomQuantities Is Nothing Then gBomQuantities = New Dictionary(Of String, String)(StringComparer.OrdinalIgnoreCase)
        If gBomStructures Is Nothing Then gBomStructures = New Dictionary(Of String, BOMStructureEnum)(StringComparer.OrdinalIgnoreCase)
    End Try
End Sub

Private Function BomStructureToDisplay(value As BOMStructureEnum) As String
    Select Case value
        Case BOMStructureEnum.kPhantomBOMStructure : Return "Phantom"
        Case BOMStructureEnum.kReferenceBOMStructure : Return "Reference"
        Case BOMStructureEnum.kPurchasedBOMStructure : Return "Purchased"
        Case BOMStructureEnum.kInseparableBOMStructure : Return "Inseparable"
        Case Else : Return "Normal"
    End Select
End Function

Private Function DisplayToBomStructure(value As String) As BOMStructureEnum
    Select Case value
        Case "Phantom" : Return BOMStructureEnum.kPhantomBOMStructure
        Case "Reference" : Return BOMStructureEnum.kReferenceBOMStructure
        Case "Purchased" : Return BOMStructureEnum.kPurchasedBOMStructure
        Case "Inseparable" : Return BOMStructureEnum.kInseparableBOMStructure
        Case Else : Return BOMStructureEnum.kNormalBOMStructure
    End Select
End Function

Private Function GetPartThumbnail(partDoc As PartDocument) As Object
    Try
        Dim picture As Object = partDoc.Thumbnail
        If picture Is Nothing Then Return Nothing
        If gPictureConversionMethod Is Nothing Then
            Dim flags As System.Reflection.BindingFlags = System.Reflection.BindingFlags.Static Or System.Reflection.BindingFlags.NonPublic
            gPictureConversionMethod = GetType(System.Windows.Forms.AxHost).GetMethod("GetPictureFromIPictureDisp", flags)
        End If
        If gPictureConversionMethod Is Nothing Then Return Nothing
        Return gPictureConversionMethod.Invoke(Nothing, New Object() {picture})
    Catch
        Return Nothing
    End Try
End Function

Private Function GetPartOccurrenceCount(filePath As String) As Integer
    If String.IsNullOrEmpty(filePath) OrElse gAsmDoc Is Nothing Then Return 1
    Dim cachedCount As Integer = 0
    If gPartCounts IsNot Nothing AndAlso gPartCounts.TryGetValue(filePath, cachedCount) Then Return cachedCount
    Dim count As Integer = 0
    Try
        For Each oOcc As ComponentOccurrence In gAsmDoc.ComponentDefinition.Occurrences.AllLeafOccurrences
            If oOcc.Suppressed Then Continue For
            If oOcc.DefinitionDocumentType = DocumentTypeEnum.kPartDocumentObject Then
                Dim partDoc As PartDocument = TryCast(oOcc.Definition.Document, PartDocument)
                If partDoc IsNot Nothing AndAlso String.Equals(partDoc.FullFileName, filePath, StringComparison.OrdinalIgnoreCase) Then
                    count += 1
                End If
            End If
        Next
    Catch
    End Try
    If count <= 0 Then Return 1
    Return count
End Function


Private Sub ApplyRowColor(rowIndex As Integer)
    If gMatGrid Is Nothing OrElse rowIndex < 0 OrElse rowIndex >= gMatGrid.Rows.Count Then Return
    Dim row As DataGridViewRow = gMatGrid.Rows(rowIndex)
    Dim status As String = ValueToText(row.Cells("Status").Value)
    Dim targetColor As System.Drawing.Color = If(rowIndex Mod 2 = 0, cRowWhite, cRowAlt)

    If status.Contains("DXF Hazır") OrElse status.Contains("/ Hazır") OrElse status = "Sac Metal" Then
        targetColor = System.Drawing.Color.FromArgb(220, 252, 231)
    ElseIf status.Contains("Dönüştürülecek") OrElse status.Contains("Adayı") OrElse status.Contains("Flat Pattern") Then
        targetColor = System.Drawing.Color.FromArgb(255, 247, 214)
    ElseIf status.Contains("Hata") OrElse status.Contains("Eksik") Then
        targetColor = System.Drawing.Color.FromArgb(254, 226, 226)
    ElseIf status.Contains("Çıktı Dışı") Then
        targetColor = System.Drawing.Color.FromArgb(226, 232, 240)
    End If

    For Each name As String In New String() {"No", "FileName", "PartName", "Description", "StockNumber", "ItemQuantity", "Status"}
        Try : row.Cells(name).Style.BackColor = targetColor : Catch : End Try
    Next
    row.Cells("Material").Style.BackColor = cInputBg
    row.Cells("BOMStructure").Style.BackColor = cInputBg
    row.Cells("PartName").Style.BackColor = cInputBg
    row.Cells("Description").Style.BackColor = cInputBg
    row.Cells("StockNumber").Style.BackColor = cInputBg
    row.Cells("Thickness").Style.BackColor = cInputBg
    row.Cells("ProductionType").Style.BackColor = cInputBg
End Sub

Private Function RowNeedsApply(row As DataGridViewRow) As Boolean
    If row Is Nothing Then Return False
    If RowHasPendingChanges(row) Then Return True
    If GetRowProductionTypeCode(row) <> PT_SHEETMETAL Then Return False
    Dim pDoc As PartDocument = Nothing
    If gUniqueParts Is Nothing OrElse Not gUniqueParts.TryGetValue(ValueToText(row.Cells("FilePath").Value), pDoc) Then Return True
    If Not IsSheetMetalPart(pDoc) Then Return True
    Try
        Dim smDef As SheetMetalComponentDefinition = TryCast(pDoc.ComponentDefinition, SheetMetalComponentDefinition)
        Return smDef Is Nothing OrElse Not smDef.HasFlatPattern
    Catch
        Return True
    End Try
End Function

Private Sub Btn_ApplyMaterials_Click(sender As Object, e As EventArgs)
    If gMatGrid Is Nothing OrElse Not gMatGrid.EndEdit() Then Return
    System.Windows.Forms.Cursor.Current = System.Windows.Forms.Cursors.WaitCursor
    gBtnApplyMaterials.Enabled = False
    Dim successCount As Integer = 0
    Dim issueCount As Integer = 0
    Dim skippedCount As Integer = 0

    Try
        For Each row As DataGridViewRow In gMatGrid.Rows
            If Not RowNeedsApply(row) Then
                skippedCount += 1
                Continue For
            End If
            Dim detail As String = ""
            If ApplyPartRowChanges(row, detail) Then
                successCount += 1
                If detail.StartsWith("UYARI:") Then AppendLog("[UYARI] " & ValueToText(row.Cells("FileName").Value) & " - " & detail.Substring(6).Trim())
            Else
                issueCount += 1
                AppendLog("[HATA] " & ValueToText(row.Cells("FileName").Value) & " - " & detail)
            End If
        Next

        If successCount > 0 Then gAsmDoc.Update()
        If issueCount = 0 Then
            PopulateMaterialGrid()
        Else
            gBomGridHasPendingEdits = True
        End If
        PopulatePartsList()
        AppendLog("[BİLGİ] Part Edit uygulandı. İşlenen: " & successCount & " | Atlanan: " & skippedCount & " | Sorun: " & issueCount)
    Catch ex As Exception
        AppendLog("[HATA] Parça güncelleme durduruldu: " & ex.Message)
        MessageBox.Show(ex.Message, "Güncelleme hatası", MessageBoxButtons.OK, MessageBoxIcon.Error)
        Return
    Finally
        gBtnApplyMaterials.Enabled = True
        System.Windows.Forms.Cursor.Current = System.Windows.Forms.Cursors.Default
    End Try

    MessageBox.Show("İşlenen parça: " & successCount & vbCrLf & _
                    "Değişmeyen parça: " & skippedCount & vbCrLf & _
                    "Sorun bulunan: " & issueCount & vbCrLf & vbCrLf & _
                    "Geçerli kalınlık girilen parçalar otomatik olarak Sac Metal'e dönüştürülür ve açınımları hazırlanır. Başarılı parçalar DXF üretimine hazırdır.", _
                    "Değişiklikler Uygulandı", MessageBoxButtons.OK, If(issueCount > 0, MessageBoxIcon.Warning, MessageBoxIcon.Information))
End Sub

Private Sub Btn_MaterialExport_Click(sender As Object, e As EventArgs)
    If gMatGrid Is Nothing Then Return
    If Not gMatGrid.EndEdit() Then Return
    Dim targetDir As String = GetDefaultExportFolder()
    Try
        If Not System.IO.Directory.Exists(targetDir) Then System.IO.Directory.CreateDirectory(targetDir)
    Catch ex As Exception
        MessageBox.Show("Export klasörü oluşturulamadı: " & ex.Message, "Export Hatası", MessageBoxButtons.OK, MessageBoxIcon.Error)
        Return
    End Try

    Dim selectedRows As New List(Of DataGridViewRow)
    For Each row As DataGridViewRow In gMatGrid.Rows
        Dim selected As Boolean = False
        Try : selected = Convert.ToBoolean(row.Cells("MaterialExportSelected").Value) : Catch : End Try
        If selected AndAlso GetRowProductionTypeCode(row) <> PT_REFERENCE Then selectedRows.Add(row)
    Next
    If selectedRows.Count = 0 Then
        MessageBox.Show("Lütfen en az bir parça seçin.", "Seçim Yapılmadı", MessageBoxButtons.OK, MessageBoxIcon.Warning)
        Return
    End If

    Dim successCount As Integer = 0
    Dim skippedCount As Integer = 0
    Dim sequence As Integer = 0
    For Each row As DataGridViewRow In selectedRows
        sequence += 1
        Dim filePath As String = CStr(row.Cells("FilePath").Value)
        Dim partName As String = CStr(row.Cells("PartName").Value)
        Dim formatName As String = CStr(row.Cells("MaterialExportFormat").Value)
        Dim extension As String = GetExportExtension(formatName)
        Dim sourceDoc As Document = Nothing
        Dim openedHere As Boolean = False
        Try
            sourceDoc = GetOpenDocumentByPath(filePath)
            If sourceDoc Is Nothing Then
                sourceDoc = ThisApplication.Documents.Open(filePath, False)
                openedHere = True
            End If
            Dim outputName As String = GetStandardExportBaseName(sourceDoc, filePath, sequence, GetPartOccurrenceCount(filePath)) & extension
            sourceDoc.SaveAs(GetAvailableOutputPath(targetDir, outputName), True)
            successCount += 1
            AppendLog("[BAŞARILI] Format Export: " & partName & " -> " & extension)
        Catch ex As Exception
            skippedCount += 1
            AppendLog("[HATA] Format Export: " & partName & " - " & ex.Message)
        Finally
            If openedHere AndAlso sourceDoc IsNot Nothing Then
                Try : sourceDoc.Close(True) : Catch : End Try
            End If
        End Try
    Next
    AppendLog("===== FORMAT EXPORT TAMAMLANDI: " & successCount & " Başarılı, " & skippedCount & " Atlandı =====")
    MessageBox.Show(successCount & " dosya dışa aktarıldı." & vbCrLf & "Klasör: " & targetDir, "Export Tamamlandı", MessageBoxButtons.OK, MessageBoxIcon.Information)
End Sub

Private Function GetSettingsFilePath() As String
    Return System.IO.Path.Combine(System.Environment.GetFolderPath(System.Environment.SpecialFolder.ApplicationData), "GokhanInventor", "SheetMetalTools.template")
End Function

Private Function LoadSavedTemplatePath() As String
    Dim settingsFile As String = GetSettingsFilePath()
    Try
        If System.IO.File.Exists(settingsFile) Then
            Dim savedPath As String = System.IO.File.ReadAllText(settingsFile).Trim()
            If System.IO.File.Exists(savedPath) Then Return savedPath
        End If
    Catch
    End Try
    Return DEFAULT_TEMPLATE_PATH
End Function

Private Sub SaveTemplatePath(templatePath As String)
    Try
        Dim settingsFile As String = GetSettingsFilePath()
        Dim settingsFolder As String = System.IO.Path.GetDirectoryName(settingsFile)
        If Not System.IO.Directory.Exists(settingsFolder) Then System.IO.Directory.CreateDirectory(settingsFolder)
        System.IO.File.WriteAllText(settingsFile, templatePath)
    Catch ex As Exception
        AppendLog("[UYARI] Şablon tercihi kaydedilemedi: " & ex.Message)
    End Try
End Sub


Private Sub PopulatePartsList()
    If gPartsList Is Nothing Then Return
    Dim hadExistingRows As Boolean = gPartsList.Rows.Count > 0
    Dim selectedBefore As New HashSet(Of String)(StringComparer.OrdinalIgnoreCase)
    For Each row As DataGridViewRow In gPartsList.Rows
        Try
            If Convert.ToBoolean(row.Cells("PartsSelected").Value) Then selectedBefore.Add(CStr(row.Tag))
        Catch
        End Try
    Next

    gAllPartsDict = GetPartsDictionary(gAsmDoc)
    gPartsListKeys = New List(Of String)
    gPartsList.SuspendLayout()
    gPartsList.Rows.Clear()

    For Each kvp As KeyValuePair(Of String, Integer) In gAllPartsDict
        Dim partDoc As PartDocument = Nothing
        If gBomParts IsNot Nothing Then gBomParts.TryGetValue(kvp.Key, partDoc)
        Dim materialName As String = ""
        Try
            If partDoc IsNot Nothing Then materialName = partDoc.ActiveMaterial.DisplayName
        Catch
        End Try
        Dim thickness As String = GetPartThicknessText(partDoc)
        Dim status As String = GetPartStatus(partDoc, GetProductionType(partDoc), thickness)
        Dim stockNumber As String = GetDesignTrackingProperty(partDoc, "Stock Number", "")
        Dim selected As Boolean
        If gSelectedForDxf IsNot Nothing AndAlso gSelectedForDxf.Count > 0 Then
            selected = gSelectedForDxf.Contains(kvp.Key)
        ElseIf hadExistingRows Then
            selected = selectedBefore.Contains(kvp.Key)
        Else
            selected = True
        End If

        Dim rowIndex As Integer = gPartsList.Rows.Add(gPartsList.Rows.Count + 1, selected, System.IO.Path.GetFileName(kvp.Key), materialName, thickness, kvp.Value, status, stockNumber)
        gPartsList.Rows(rowIndex).Tag = kvp.Key
        gPartsListKeys.Add(kvp.Key)
        If status.Contains("DXF Hazır") Then
            gPartsList.Rows(rowIndex).DefaultCellStyle.BackColor = System.Drawing.Color.FromArgb(220, 252, 231)
        ElseIf status.Contains("Açınım") Then
            gPartsList.Rows(rowIndex).DefaultCellStyle.BackColor = System.Drawing.Color.FromArgb(255, 247, 214)
        End If
    Next
    gPartsList.ResumeLayout()
    AutoSizeDxfLeftPanel(Nothing, EventArgs.Empty)
End Sub

Private Function GetPartThicknessText(partDoc As PartDocument) As String
    If partDoc Is Nothing Then Return ""
    Try
        Dim smDef As SheetMetalComponentDefinition = TryCast(partDoc.ComponentDefinition, SheetMetalComponentDefinition)
        If smDef Is Nothing Then Return ""
        Dim thicknessMm As Double = GetThicknessMillimeters(partDoc, smDef)
        If thicknessMm <= 0 Then Return ""
        Return thicknessMm.ToString("0.##", System.Globalization.CultureInfo.InvariantCulture) & " mm"
    Catch
        Try
            Dim smDef As SheetMetalComponentDefinition = TryCast(partDoc.ComponentDefinition, SheetMetalComponentDefinition)
            If smDef Is Nothing Then Return ""
            Return smDef.Thickness.Expression
        Catch
            Return ""
        End Try
    End Try
End Function

Private Function GetThicknessMillimeters(partDoc As PartDocument, smDef As SheetMetalComponentDefinition) As Double
    Return CDbl(partDoc.UnitsOfMeasure.ConvertUnits(smDef.Thickness.Value, _
        UnitsTypeEnum.kDatabaseLengthUnits, UnitsTypeEnum.kMillimeterLengthUnits))
End Function

Private Sub CreatePartsXlsx(filePath As String)
    Dim contentTypes As String = "<?xml version=""1.0"" encoding=""UTF-8"" standalone=""yes""?>" & _
        "<Types xmlns=""http://schemas.openxmlformats.org/package/2006/content-types"">" & _
        "<Default Extension=""rels"" ContentType=""application/vnd.openxmlformats-package.relationships+xml""/>" & _
        "<Default Extension=""xml"" ContentType=""application/xml""/>" & _
        "<Override PartName=""/xl/workbook.xml"" ContentType=""application/vnd.openxmlformats-officedocument.spreadsheetml.sheet.main+xml""/>" & _
        "<Override PartName=""/xl/worksheets/sheet1.xml"" ContentType=""application/vnd.openxmlformats-officedocument.spreadsheetml.worksheet+xml""/>" & _
        "</Types>"
    Dim rootRels As String = "<?xml version=""1.0"" encoding=""UTF-8"" standalone=""yes""?>" & _
        "<Relationships xmlns=""http://schemas.openxmlformats.org/package/2006/relationships"">" & _
        "<Relationship Id=""rId1"" Type=""http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument"" Target=""xl/workbook.xml""/>" & _
        "</Relationships>"
    Dim workbook As String = "<?xml version=""1.0"" encoding=""UTF-8"" standalone=""yes""?>" & _
        "<workbook xmlns=""http://schemas.openxmlformats.org/spreadsheetml/2006/main"" xmlns:r=""http://schemas.openxmlformats.org/officeDocument/2006/relationships"">" & _
        "<sheets><sheet name=""Sac Parça Listesi"" sheetId=""1"" r:id=""rId1""/></sheets></workbook>"
    Dim workbookRels As String = "<?xml version=""1.0"" encoding=""UTF-8"" standalone=""yes""?>" & _
        "<Relationships xmlns=""http://schemas.openxmlformats.org/package/2006/relationships"">" & _
        "<Relationship Id=""rId1"" Type=""http://schemas.openxmlformats.org/officeDocument/2006/relationships/worksheet"" Target=""worksheets/sheet1.xml""/>" & _
        "</Relationships>"

    Dim sheet As New StringBuilder()
    sheet.Append("<?xml version=""1.0"" encoding=""UTF-8"" standalone=""yes""?>")
    sheet.Append("<worksheet xmlns=""http://schemas.openxmlformats.org/spreadsheetml/2006/main""><sheetData>")
    AppendXlsxRow(sheet, 1, New String() {"Item NO", "Filename", "Item QTY", "Material", "Thickness", "Stock Number"})
    For i As Integer = 0 To gPartsList.Rows.Count - 1
        AppendXlsxRow(sheet, i + 2, New String() { _
            ValueToText(gPartsList.Rows(i).Cells("PartsNo").Value), _
            ValueToText(gPartsList.Rows(i).Cells("PartsFileName").Value), _
            ValueToText(gPartsList.Rows(i).Cells("PartsQuantity").Value), _
            ValueToText(gPartsList.Rows(i).Cells("PartsMaterial").Value), _
            ValueToText(gPartsList.Rows(i).Cells("PartsThickness").Value), _
            ValueToText(gPartsList.Rows(i).Cells("PartsStockNumber").Value)})
    Next
    sheet.Append("</sheetData></worksheet>")

    Using package As New System.IO.Compression.ZipArchive( _
        New System.IO.FileStream(filePath, System.IO.FileMode.Create), _
        System.IO.Compression.ZipArchiveMode.Create)
        WriteXlsxEntry(package, "[Content_Types].xml", contentTypes)
        WriteXlsxEntry(package, "_rels/.rels", rootRels)
        WriteXlsxEntry(package, "xl/workbook.xml", workbook)
        WriteXlsxEntry(package, "xl/_rels/workbook.xml.rels", workbookRels)
        WriteXlsxEntry(package, "xl/worksheets/sheet1.xml", sheet.ToString())
    End Using
End Sub

Private Sub AppendXlsxRow(builder As StringBuilder, rowNumber As Integer, values As String())
    builder.Append("<row r=""" & rowNumber.ToString() & """>")
    For i As Integer = 0 To values.Length - 1
        Dim cellRef As String = Chr(65 + i).ToString() & rowNumber.ToString()
        builder.Append("<c r=""" & cellRef & """ t=""inlineStr""><is><t>")
        builder.Append(System.Security.SecurityElement.Escape(If(values(i), "")))
        builder.Append("</t></is></c>")
    Next
    builder.Append("</row>")
End Sub

Private Sub WriteXlsxEntry(package As System.IO.Compression.ZipArchive, entryName As String, content As String)
    Dim entry As System.IO.Compression.ZipArchiveEntry = package.CreateEntry(entryName)
    Using stream As System.IO.Stream = entry.Open()
        Using writer As New System.IO.StreamWriter(stream, New System.Text.UTF8Encoding(False))
            writer.Write(content)
        End Using
    End Using
End Sub

Private Function GetAvailableOutputPath(folder As String, fileName As String) As String
    Dim candidate As String = System.IO.Path.Combine(folder, fileName)
    If Not System.IO.File.Exists(candidate) Then Return candidate
    Dim baseName As String = System.IO.Path.GetFileNameWithoutExtension(fileName)
    Dim extension As String = System.IO.Path.GetExtension(fileName)
    Dim suffix As Integer = 2
    Do
        candidate = System.IO.Path.Combine(folder, baseName & "_" & suffix.ToString() & extension)
        suffix += 1
    Loop While System.IO.File.Exists(candidate)
    Return candidate
End Function

Private Sub PopulateExportGrid()
    If gExportGrid Is Nothing Then Return
    Dim hadExistingRows As Boolean = gExportGrid.Rows.Count > 0
    Dim selectedBefore As New HashSet(Of String)(StringComparer.OrdinalIgnoreCase)
    Dim formatBefore As New Dictionary(Of String, String)(StringComparer.OrdinalIgnoreCase)
    For Each oldRow As DataGridViewRow In gExportGrid.Rows
        Dim oldPath As String = ValueToText(oldRow.Cells("ExportFilePath").Value)
        If String.IsNullOrEmpty(oldPath) Then Continue For
        Dim selected As Boolean = False
        Try : selected = Convert.ToBoolean(oldRow.Cells("ExportSelected").Value) : Catch : End Try
        If selected Then selectedBefore.Add(oldPath)
        formatBefore(oldPath) = ValueToText(oldRow.Cells("ExportFormat").Value)
    Next
    gExportParts = New Dictionary(Of String, Document)
    gExportPartCounts = New Dictionary(Of String, Integer)
    ReadBomQuantities()
    For Each kvp As KeyValuePair(Of String, PartDocument) In gBomParts
        gExportParts.Add(kvp.Key, kvp.Value)
        Dim qtyText As String = Nothing
        Dim qty As Integer = 1
        If gBomQuantities.TryGetValue(kvp.Key, qtyText) Then Integer.TryParse(qtyText, qty)
        gExportPartCounts.Add(kvp.Key, Math.Max(1, qty))
    Next

    gExportGrid.SuspendLayout()
    gExportGrid.Rows.Clear()
    For Each kvp As KeyValuePair(Of String, Document) In gExportParts
        Dim filePath As String = kvp.Key
        Dim oDoc As Document = kvp.Value
        Dim description As String = ""
        Try : description = CStr(oDoc.PropertySets.Item("Design Tracking Properties").Item("Description").Value) : Catch : End Try
        Dim stockNumber As String = ""
        Try : stockNumber = CStr(oDoc.PropertySets.Item("Design Tracking Properties").Item("Stock Number").Value) : Catch : End Try
        Dim displayName As String = GetDesignTrackingProperty(oDoc, "Part Number", System.IO.Path.GetFileNameWithoutExtension(filePath))
        Dim savedFormat As String = Nothing
        If Not formatBefore.TryGetValue(filePath, savedFormat) OrElse System.Array.IndexOf(gExportFormatItems, savedFormat) < 0 Then savedFormat = gExportFormatItems(0)
        Dim rowIndex As Integer = gExportGrid.Rows.Add(If(hadExistingRows, selectedBefore.Contains(filePath), True), gExportGrid.Rows.Count + 1, filePath, displayName, description, gExportPartCounts(filePath), savedFormat)
        gExportGrid.Rows(rowIndex).Cells("ExportFileName").Value = System.IO.Path.GetFileName(filePath)
        gExportGrid.Rows(rowIndex).Cells("ExportStockNumber").Value = stockNumber
        gExportGrid.Rows(rowIndex).Cells("ExportSelected").Style.BackColor = cRowFilled
    Next
    gExportGrid.ResumeLayout()
    FreezeGridColumnsToContent(gExportGrid, 55, 420)
End Sub

Private Function GetExportExtension(formatName As String) As String
    If formatName.Contains(".stp") Then Return ".stp"
    If formatName.Contains(".igs") Then Return ".igs"
    If formatName.Contains(".sat") Then Return ".sat"
    If formatName.Contains(".stl") Then Return ".stl"
    If formatName.Contains(".dwg") Then Return ".dwg"
    If formatName.Contains(".dxf") Then Return ".dxf"
    If formatName.Contains(".jt") Then Return ".jt"
    If formatName.Contains(".obj") Then Return ".obj"
    If formatName.Contains(".x_t") Then Return ".x_t"
    If formatName.Contains(".x_b") Then Return ".x_b"
    If formatName.Contains(".glb") Then Return ".glb"
    Return ".stp"
End Function

Private Sub Btn_ExportSelected_Click(sender As Object, e As EventArgs)
    gExportGrid.EndEdit()
    Dim targetDir As String = gTxtExportPath.Text.Trim()
    If String.IsNullOrEmpty(targetDir) Then
        MessageBox.Show("Lütfen dışa aktarma konumu seçin.", "Konum Eksik", MessageBoxButtons.OK, MessageBoxIcon.Warning)
        Return
    End If

    Dim selectedRows As New List(Of DataGridViewRow)
    For Each row As DataGridViewRow In gExportGrid.Rows
        If CBool(row.Cells("ExportSelected").Value) Then selectedRows.Add(row)
    Next
    If selectedRows.Count = 0 Then
        MessageBox.Show("Lütfen en az bir parça veya alt montaj seçin.", "Seçim Yapılmadı", MessageBoxButtons.OK, MessageBoxIcon.Warning)
        Return
    End If

    Try
        If Not System.IO.Directory.Exists(targetDir) Then System.IO.Directory.CreateDirectory(targetDir)
    Catch ex As Exception
        MessageBox.Show("Export klasörü oluşturulamadı: " & ex.Message, "Hata", MessageBoxButtons.OK, MessageBoxIcon.Error)
        Return
    End Try

    SetButtonsEnabled(False)
    gBtnMaterialExport.Enabled = False
    gBtnMaterialRefresh.Enabled = False
    gExportGrid.Enabled = False
    gBtnExportSelectAll.Enabled = False
    gBtnExportSelectNone.Enabled = False
    gCancelRequested = False
    If gBtnCancel IsNot Nothing Then gBtnCancel.Enabled = True
    Dim successCount As Integer = 0
    Dim skippedCount As Integer = 0
    Dim sequence As Integer = 0
    Dim processedCount As Integer = 0
    If gProgressBar IsNot Nothing Then gProgressBar.Value = 0
    AppendLog("===== TOPLU EXPORT BAŞLATILDI (" & selectedRows.Count & " Dosya) =====")

    Try
        For Each row As DataGridViewRow In selectedRows
            If gCancelRequested Then Exit For
            sequence += 1
            processedCount += 1
            UpdateProgress(processedCount, selectedRows.Count)
            Dim filePath As String = CStr(row.Cells("ExportFilePath").Value)
            Dim partName As String = CStr(row.Cells("ExportPartName").Value)
            Dim formatName As String = CStr(row.Cells("ExportFormat").Value)
            Dim extension As String = GetExportExtension(formatName)
            Dim qty As Integer = CInt(row.Cells("ExportQuantity").Value)
            Dim openedHere As Boolean = False
            Dim sourceDoc As Document = Nothing

            Try
                sourceDoc = GetOpenDocumentByPath(filePath)
                If sourceDoc Is Nothing Then
                    sourceDoc = ThisApplication.Documents.Open(filePath, False)
                    openedHere = True
                End If
                Dim outputName As String = GetStandardExportBaseName(sourceDoc, filePath, sequence, qty) & extension
                Dim outputPath As String = GetAvailableOutputPath(targetDir, outputName)
                sourceDoc.SaveAs(outputPath, True)
                successCount += 1
                AppendLog("   [BAŞARILI] " & partName & " -> " & extension)
            Catch ex As Exception
                skippedCount += 1
                AppendLog("   [HATA] " & partName & " (" & extension & "): " & ex.Message)
            Finally
                If openedHere AndAlso sourceDoc IsNot Nothing Then
                    Try : sourceDoc.Close(True) : Catch : End Try
                End If
            End Try
            System.Windows.Forms.Application.DoEvents()
        Next
    Finally
        SetButtonsEnabled(True)
        gBtnMaterialExport.Enabled = True
        gBtnMaterialRefresh.Enabled = True
        gExportGrid.Enabled = True
        gBtnExportSelected.Enabled = True
        gBtnExportSelectAll.Enabled = True
        gBtnExportSelectNone.Enabled = True
        If gBtnCancel IsNot Nothing Then gBtnCancel.Enabled = False
    End Try

    If gCancelRequested Then AppendLog("[BİLGİ] Export işlemi iptal edildi.")
    AppendLog("===== EXPORT TAMAMLANDI: " & successCount & " Başarılı, " & skippedCount & " Atlandı =====")
    Try : System.Diagnostics.Process.Start("explorer.exe", """" & targetDir & """") : Catch : End Try
End Sub

Private Function GetOpenDocumentByPath(filePath As String) As Document
    For Each oDoc As Document In ThisApplication.Documents
        Try
            If String.Equals(oDoc.FullFileName, filePath, StringComparison.OrdinalIgnoreCase) Then Return oDoc
        Catch
        End Try
    Next
    Return Nothing
End Function

Private Sub CloseProcessedPartTab(partDoc As PartDocument, openedHere As Boolean)
    If partDoc Is Nothing Then Return
    Try
        ' Montajdaki görünmez referansları kapatmaya gerek yoktur; yalnızca açık sekmeleri kapat.
        If partDoc.Views.Count = 0 Then
            If openedHere Then partDoc.Close(True)
            Return
        End If
        ' Sekmeyi kapatmadan önce açınım ve kalınlık değişikliklerini koru.
        If partDoc.Dirty Then partDoc.Save()
        partDoc.Close(True)
    Catch ex As Exception
        AppendLog("   [UYARI] Parça sekmesi kapatılamadı: " & ex.Message)
    End Try
End Sub

Private Function IsSheetMetalPart(pDoc As PartDocument) As Boolean
    If pDoc Is Nothing Then Return False
    Try
        Return pDoc.SubType = SHEETMETAL_SUBTYPE OrElse TypeOf pDoc.ComponentDefinition Is SheetMetalComponentDefinition
    Catch
        Return False
    End Try
End Function

Private Function NormalizeProductionType(value As String) As String
    If String.IsNullOrWhiteSpace(value) Then Return ""
    Dim s As String = value.Trim().ToLowerInvariant()
    Select Case s
        Case "sheetmetal", "sheet metal", "sac metal", "sacmetal" : Return PT_SHEETMETAL
        Case "machining", "talaşlı imalat", "talasli imalat", "talaşlı", "talasli" : Return PT_MACHINING
        Case "fastener", "bağlantı malzemeleri", "baglanti malzemeleri", "bağlantı", "baglanti" : Return PT_FASTENER
        Case "purchased", "hazır malzeme", "hazir malzeme", "satın alma", "satin alma" : Return PT_PURCHASED
        Case "reference", "referans" : Return PT_REFERENCE
        Case "tanımsız", "tanimsiz", "undefined", "none" : Return ""
    End Select
    Return ""
End Function

Private Function ProductionTypeToDisplay(code As String) As String
    Select Case NormalizeProductionType(code)
        Case PT_SHEETMETAL : Return "Sac Metal"
        Case PT_MACHINING : Return "Talaşlı İmalat"
        Case PT_FASTENER : Return "Bağlantı Malzemeleri"
        Case PT_PURCHASED : Return "Hazır Malzeme"
        Case PT_REFERENCE : Return "Referans"
        Case Else : Return "Tanımsız"
    End Select
End Function

Private Function IsLikelyFastener(pDoc As PartDocument) As Boolean
    If pDoc Is Nothing Then Return False
    Try
        Dim defObj As Object = pDoc.ComponentDefinition
        Dim isContent As Boolean = False
        Try : isContent = CBool(defObj.IsContentMember) : Catch : End Try
        If Not isContent Then Return False
        Dim probe As String = (System.IO.Path.GetFileNameWithoutExtension(pDoc.FullFileName) & " " & _
                               GetDesignTrackingProperty(pDoc, "Description", "") & " " & _
                               GetDesignTrackingProperty(pDoc, "Part Number", "")).ToLowerInvariant()
        Dim keys As String() = {"cıvata", "civata", "bolt", "screw", "vida", "somun", "nut", "washer", "rondela", "pul", "pin", "pim", "segman", "retaining ring"}
        For Each k As String In keys
            If probe.Contains(k) Then Return True
        Next
    Catch
    End Try
    Return False
End Function

Private Function TryParseThickness(thicknessText As String, ByRef thicknessMm As Double) As Boolean
    thicknessMm = 0
    If String.IsNullOrWhiteSpace(thicknessText) Then Return False
    Dim clean As String = thicknessText.Trim().ToLowerInvariant().Replace("mm", "").Trim().Replace(",", ".")
    If Not Double.TryParse(clean, System.Globalization.NumberStyles.Any, System.Globalization.CultureInfo.InvariantCulture, thicknessMm) Then Return False
    Return thicknessMm > 0 AndAlso Not Double.IsNaN(thicknessMm) AndAlso Not Double.IsInfinity(thicknessMm)
End Function

Private Function CanConvertToSheetMetal(pDoc As PartDocument) As Boolean
    If pDoc Is Nothing Then Return False
    If IsSheetMetalPart(pDoc) Then Return True
    Try
        Return pDoc.ComponentDefinition.SurfaceBodies.Count > 0
    Catch
        Return False
    End Try
End Function

Private Function ApplyMaterialToPart(pDoc As PartDocument, materialName As String, ByRef errorText As String) As Boolean
    errorText = ""
    If pDoc Is Nothing OrElse String.IsNullOrWhiteSpace(materialName) Then Return True
    Try
        pDoc.ActiveMaterial = pDoc.MaterialAssets.Item(materialName)
        Return True
    Catch
    End Try
    Try
        For Each asset As MaterialAsset In pDoc.MaterialAssets
            If String.Equals(asset.DisplayName, materialName, StringComparison.OrdinalIgnoreCase) Then
                pDoc.ActiveMaterial = asset
                Return True
            End If
        Next
        If gMatLib IsNot Nothing Then
            For Each asset As MaterialAsset In gMatLib.MaterialAssets
                If String.Equals(asset.DisplayName, materialName, StringComparison.OrdinalIgnoreCase) Then
                    pDoc.ActiveMaterial = asset.CopyTo(pDoc)
                    Return True
                End If
            Next
        End If
        errorText = "Malzeme bulunamadı: " & materialName
    Catch ex As Exception
        errorText = "Malzeme uygulanamadı: " & ex.Message
    End Try
    Return False
End Function

Private Function ApplySheetMetalSettings(pDoc As PartDocument, materialName As String, thicknessText As String, ByRef errorText As String) As Boolean
    errorText = ""
    If pDoc Is Nothing Then
        errorText = "Parça dokümanı bulunamadı."
        Return False
    End If
    If Not CanConvertToSheetMetal(pDoc) Then
        errorText = "Geometri Sac Metal dönüşümü için uygun bir katı gövde içermiyor."
        Return False
    End If

    Dim thicknessMm As Double = 0
    If Not TryParseThickness(thicknessText, thicknessMm) Then
        errorText = "Sac Metal dönüşümü için 0'dan büyük geçerli bir kalınlık girilmelidir."
        Return False
    End If

    Dim materialError As String = ""
    If Not ApplyMaterialToPart(pDoc, materialName, materialError) Then
        errorText = materialError
        Return False
    End If

    If Not IsSheetMetalPart(pDoc) Then
        Try
            pDoc.SubType = SHEETMETAL_SUBTYPE
        Catch ex As Exception
            errorText = "Sac metale dönüşüm başarısız: " & ex.Message
            Return False
        End Try
    End If

    Dim smDef As SheetMetalComponentDefinition = TryCast(pDoc.ComponentDefinition, SheetMetalComponentDefinition)
    If smDef Is Nothing Then
        errorText = "SheetMetalComponentDefinition alınamadı."
        Return False
    End If

    Try
        smDef.UseSheetMetalStyleThickness = False
        smDef.Thickness.Expression = thicknessMm.ToString(System.Globalization.CultureInfo.InvariantCulture) & " mm"
    Catch ex As Exception
        errorText = "Kalınlık uygulanamadı: " & ex.Message
        Return False
    End Try

    Try
        Dim oUnfold As UnfoldMethod = smDef.UnfoldMethods.Item("Default_KFactor")
        smDef.UnfoldMethod = oUnfold
    Catch
    End Try
    Try : pDoc.Update() : Catch : End Try
    Return True
End Function

Private Function TryCreateFlatPattern(pDoc As PartDocument, ByRef errorText As String) As Boolean
    errorText = ""
    If pDoc Is Nothing OrElse Not IsSheetMetalPart(pDoc) Then
        errorText = "Parça Sac Metal değil."
        Return False
    End If
    Try
        Dim smDef As SheetMetalComponentDefinition = TryCast(pDoc.ComponentDefinition, SheetMetalComponentDefinition)
        If smDef Is Nothing Then
        errorText = "SheetMetalComponentDefinition alınamadı."
        Return False
    End If
        If smDef.HasFlatPattern Then Return True
        smDef.Unfold()
        smDef.FlatPattern.ExitEdit()
        Try : pDoc.Update() : Catch : End Try
        Return smDef.HasFlatPattern
    Catch ex As Exception
        errorText = ex.Message
        Return False
    End Try
End Function

Private Function GetPartStatus(pDoc As PartDocument, productionType As String, thicknessText As String) As String
    If pDoc Is Nothing Then Return "Hata: Doküman Yok"
    Dim pType As String = NormalizeProductionType(productionType)
    If pType = PT_REFERENCE Then Return "Çıktı Dışı"
    If pType = PT_FASTENER Then Return "Bağlantı / Hazır"
    If pType = PT_PURCHASED Then Return "Satın Alma / Hazır"
    If pType = PT_MACHINING Then Return "Talaşlı / Hazır"

    If pType = PT_SHEETMETAL Then
        If Not IsSheetMetalPart(pDoc) Then
            Dim t As Double = 0
            If TryParseThickness(thicknessText, t) Then Return "Dönüştürülecek"
            Return "Sac Metal Adayı"
        End If
        Try
            Dim smDef As SheetMetalComponentDefinition = TryCast(pDoc.ComponentDefinition, SheetMetalComponentDefinition)
            If smDef Is Nothing Then Return "Hata: Sac Metal"
            If Not smDef.HasFlatPattern Then Return "Açınım Yok"
            Return "DXF Hazır"
        Catch
            Return "Hata: Sac Metal"
        End Try
    End If

    If IsSheetMetalPart(pDoc) Then Return "Sac Metal"
    Return "Normal Part"
End Function

Private Function ApplyPartRowChanges(row As DataGridViewRow, ByRef detail As String) As Boolean
    detail = ""
    If row Is Nothing Then
        detail = "Satır bulunamadı."
        Return False
    End If
    Dim filePath As String = ValueToText(row.Cells("FilePath").Value)
    Dim pDoc As PartDocument = Nothing
    If gUniqueParts Is Nothing OrElse Not gUniqueParts.TryGetValue(filePath, pDoc) Then
        detail = "Parça dokümanı bulunamadı."
        Return False
    End If

    Dim materialName As String = ValueToText(row.Cells("Material").Value)
    Dim thicknessText As String = ValueToText(row.Cells("Thickness").Value)
    Dim productionType As String = GetRowProductionTypeCode(row)

    ' Geçerli kalınlık girilmiş bir parça Sac Metal kabul edilir.
    ' STEP/normal IPT için ayrıca Üretim Tipi seçmek gerekmez.
    Dim enteredThickness As Double = 0
    Dim hasValidThickness As Boolean = TryParseThickness(thicknessText, enteredThickness)
    If hasValidThickness Then
        productionType = PT_SHEETMETAL
        row.Cells("ProductionType").Value = ProductionTypeToDisplay(PT_SHEETMETAL)
    End If

    If productionType = PT_SHEETMETAL AndAlso Not hasValidThickness AndAlso Not IsSheetMetalPart(pDoc) Then
        detail = "Sac Metal dönüşümü için 0'dan büyük geçerli bir kalınlık girilmelidir."
        Return False
    End If

    Try
        If BomCellWasEdited(row, "PartName") Then SetDesignTrackingPropertyIfChanged(pDoc, "Part Number", ValueToText(row.Cells("PartName").Value))
        If BomCellWasEdited(row, "Description") Then SetDesignTrackingPropertyIfChanged(pDoc, "Description", ValueToText(row.Cells("Description").Value))
        If BomCellWasEdited(row, "StockNumber") Then SetDesignTrackingPropertyIfChanged(pDoc, "Stock Number", ValueToText(row.Cells("StockNumber").Value))
        If BomCellWasEdited(row, "BOMStructure") Then
            Dim requestedStructure As BOMStructureEnum = DisplayToBomStructure(ValueToText(row.Cells("BOMStructure").Value))
            If pDoc.ComponentDefinition.BOMStructure <> requestedStructure Then
                pDoc.ComponentDefinition.BOMStructure = requestedStructure
            End If
        End If
        If productionType <> GetProductionType(pDoc) Then SetProductionType(pDoc, productionType)
    Catch ex As Exception
        detail = ex.Message
        Return False
    End Try

    Dim materialError As String = ""
    If BomCellWasEdited(row, "Material") AndAlso Not ApplyMaterialToPart(pDoc, materialName, materialError) Then
        detail = materialError
        Return False
    End If

    If productionType = PT_SHEETMETAL Then
        If hasValidThickness AndAlso (BomCellWasEdited(row, "Thickness") OrElse Not IsSheetMetalPart(pDoc)) Then
            Dim smError As String = ""
            If Not ApplySheetMetalSettings(pDoc, materialName, thicknessText, smError) Then
                detail = smError
                Return False
            End If
        End If

        If Not IsSheetMetalPart(pDoc) Then
            detail = "Parça Sac Metal'e dönüştürülemedi."
            Return False
        End If

        ' Değişiklikleri Uygula sırasında açınımı otomatik hazırla.
        Dim flatError As String = ""
        If Not TryCreateFlatPattern(pDoc, flatError) Then
            detail = "Sac Metal tanımı uygulandı ancak açınım oluşturulamadı: " & flatError
            Return False
        End If
    End If

    Try
        pDoc.Update()
    Catch ex As Exception
        detail = "Parça güncellenemedi: " & ex.Message
        Return False
    End Try
    Return True
End Function

Private Sub AppendLog(msg As String)
    If msg Is Nothing Then msg = ""
    If msg.IndexOf("[HATA]", StringComparison.OrdinalIgnoreCase) >= 0 Then gOperationErrorCount += 1
    Dim line As String = "[" & DateTime.Now.ToString("HH:mm:ss") & "] " & msg
    If gLogBox IsNot Nothing AndAlso Not gLogBox.IsDisposed Then
        gLogBox.AppendText(line & vbCrLf)
        gLogBox.SelectionStart = gLogBox.Text.Length
        gLogBox.ScrollToCaret()
    End If
    If gStatusLabel IsNot Nothing AndAlso Not gStatusLabel.IsDisposed Then
        Dim statusText As String = msg.Replace(vbCr, " ").Replace(vbLf, " ").Trim()
        If statusText.Length > 180 Then statusText = statusText.Substring(0, 180) & "..."
        gStatusLabel.Text = "Durum  |  " & statusText
    End If
    System.Windows.Forms.Application.DoEvents()
End Sub

Private Function GetPartsDictionary(oAsm As AssemblyDocument) As Dictionary(Of String, Integer)
    ReadBomQuantities()
    Dim dict As New Dictionary(Of String, Integer)(StringComparer.OrdinalIgnoreCase)
    For Each kvp As KeyValuePair(Of String, PartDocument) In gBomParts
        If Not IsSheetMetalPart(kvp.Value) OrElse GetProductionType(kvp.Value) <> PT_SHEETMETAL Then Continue For
        Dim qtyText As String = Nothing
        Dim qty As Integer = 1
        If gBomQuantities.TryGetValue(kvp.Key, qtyText) Then Integer.TryParse(qtyText, qty)
        dict.Add(kvp.Key, Math.Max(1, qty))
    Next
    Return dict
End Function

Private Function SanitizeFileName(name As String) As String
    Dim invalidChars As Char() = System.IO.Path.GetInvalidFileNameChars()
    Dim result As String = name
    For Each c As Char In invalidChars
        result = result.Replace(c, "_")
    Next
    Return result
End Function

Private Function GetStandardExportBaseName(oDoc As Document, filePath As String, sequence As Integer, quantity As Integer) As String
    Dim stockNumber As String = GetDesignTrackingProperty(oDoc, "Stock Number", "")
    If String.IsNullOrWhiteSpace(stockNumber) Then stockNumber = "STOK_NO_YOK"
    Dim materialName As String = "ASSEMBLY"
    Dim thicknessText As String = "NA"

    Dim partDoc As PartDocument = TryCast(oDoc, PartDocument)
    If partDoc IsNot Nothing Then
        Try : materialName = partDoc.ActiveMaterial.DisplayName : Catch : End Try
        Try
            If IsSheetMetalPart(partDoc) Then
                Dim smDef As SheetMetalComponentDefinition = TryCast(partDoc.ComponentDefinition, SheetMetalComponentDefinition)
                If smDef IsNot Nothing Then
                    Dim thicknessValue As Double = Math.Round(GetThicknessMillimeters(partDoc, smDef), 2)
                    thicknessText = thicknessValue.ToString("0.##", System.Globalization.CultureInfo.InvariantCulture) & "MM"
                End If
            End If
        Catch
        End Try
    End If

    If String.IsNullOrEmpty(materialName) Then materialName = "MATERIAL_NA"
    Return String.Format("{0:000}_{1}_{2}AD_{3}_{4}", sequence, SanitizeFileName(stockNumber), quantity, SanitizeFileName(materialName), thicknessText)
End Function

Private Sub DoExportDXF(ByRef successCount As Integer, ByRef skippedCount As Integer, dictParts As Dictionary(Of String, Integer), Optional sequenceStart As Integer = 0, Optional totalOverride As Integer = 0, Optional closePartTab As Boolean = True)
    Dim dxfFolder As String = GetSharedOutputFolder()
    gLastOutputFolder = dxfFolder

    Dim total As Integer = If(totalOverride > 0, totalOverride, dictParts.Count)
    Dim processed As Integer = sequenceStart
    Dim dxfSeq As Integer = sequenceStart

    Dim originalSU As Boolean = ThisApplication.ScreenUpdating
    Dim originalSilent As Boolean = ThisApplication.SilentOperation
    ThisApplication.ScreenUpdating = False
    ThisApplication.SilentOperation = True

    Try
        For Each kvp As KeyValuePair(Of String, Integer) In dictParts
            If gCancelRequested Then
                AppendLog("[BİLGİ] DXF işlemi kullanıcı tarafından durduruldu.")
                Exit For
            End If
            processed += 1
            UpdateProgress(processed, total)
            Dim filePath As String = kvp.Key
            Dim qty As Integer = kvp.Value
            Dim partName As String = System.IO.Path.GetFileNameWithoutExtension(filePath)
            If gLblDxfStatus IsNot Nothing Then gLblDxfStatus.Text = "İşlem Durumu: DXF | " & processed & " / " & total & " | " & partName
            AppendLog("[" & processed & "/" & total & "] " & partName)

            Dim oPartDoc As PartDocument = Nothing
            Dim openedHere As Boolean = False
            Try
                oPartDoc = TryCast(GetOpenDocumentByPath(filePath), PartDocument)
                If oPartDoc Is Nothing Then
                    oPartDoc = CType(ThisApplication.Documents.Open(filePath, False), PartDocument)
                    openedHere = True
                End If

                If GetProductionType(oPartDoc) <> PT_SHEETMETAL Then
                    skippedCount += 1
                    AppendLog("   [UYARI] Üretim tipi Sheet Metal değil, atlandı.")
                    Continue For
                End If

                If TypeOf oPartDoc.ComponentDefinition Is SheetMetalComponentDefinition Then
                    Dim oSMDef As SheetMetalComponentDefinition = CType(oPartDoc.ComponentDefinition, SheetMetalComponentDefinition)
                    Try : oPartDoc.Update() : Catch : End Try

                    If Not oSMDef.HasFlatPattern Then
                        Try
                            oSMDef.Unfold()
                            oSMDef.FlatPattern.ExitEdit()
                        Catch ex As Exception
                            skippedCount += 1
                            AppendLog("   [HATA] Açınım oluşturulamadı.")
                            Continue For
                        End Try
                    End If

                    Dim thkValue As Double = 0
                    Try : thkValue = GetThicknessMillimeters(oPartDoc, oSMDef) : Catch : End Try
                    If thkValue <= 0 Then
                        skippedCount += 1
                        AppendLog("   [HATA] Geçerli sac kalınlığı okunamadı; DXF atlandı.")
                        Continue For
                    End If
                    Dim thkText As String = Math.Round(thkValue, 2).ToString(System.Globalization.CultureInfo.InvariantCulture)

                    Dim targetFolder As String = System.IO.Path.Combine(dxfFolder, thkText & "MM")
                    If Not System.IO.Directory.Exists(targetFolder) Then System.IO.Directory.CreateDirectory(targetFolder)

                    dxfSeq += 1
                    Dim dxfFileName As String = GetStandardExportBaseName(oPartDoc, filePath, dxfSeq, qty) & ".dxf"
                    Dim fullSavePath As String = GetAvailableOutputPath(targetFolder, dxfFileName)
                    Dim stagingPath As String = System.IO.Path.Combine(targetFolder, "." & System.Guid.NewGuid().ToString("N") & ".pending.dxf")

                    Try
                        ExportFlatFaceAsDxf(oPartDoc, oSMDef, stagingPath)
                        Dim contourError As String = ""
                        If Not ValidateClosedDxfContours(stagingPath, contourError) Then
                            Throw New System.IO.InvalidDataException("Export Face As konturu doğrulanamadı: " & contourError)
                        End If
                        System.IO.File.Move(stagingPath, fullSavePath)
                        successCount += 1
                        AppendLog("   [BAŞARILI] Export Face As DXF yazıldı (" & thkText & "mm -> " & thkText & "MM klasörüne)")
                    Catch ex As Exception
                        Try
                            If System.IO.File.Exists(stagingPath) Then System.IO.File.Delete(stagingPath)
                        Catch cleanupEx As Exception
                            AppendLog("   [HATA] Geçici DXF silinemedi: " & stagingPath & " | " & cleanupEx.Message)
                        End Try
                        skippedCount += 1
                        AppendLog("   [HATA] DXF reddedildi/kaydedilemedi: " & ex.Message)
                    End Try
                Else
                    skippedCount += 1
                    AppendLog("   [UYARI] Sac metal değil, atlandı.")
                End If

            Catch ex As Exception
                skippedCount += 1
                AppendLog("   [HATA] Dosya açılamadı: " & ex.Message)
            Finally
                If closePartTab Then CloseProcessedPartTab(oPartDoc, openedHere)
            End Try
        Next
    Finally
        ThisApplication.ScreenUpdating = originalSU
        ThisApplication.SilentOperation = originalSilent
    End Try
End Sub

Private Sub ExportFlatFaceAsDxf(pDoc As PartDocument, smDef As SheetMetalComponentDefinition, outputPath As String)
    Dim oldScreenUpdating As Boolean = ThisApplication.ScreenUpdating
    Dim oldSilentOperation As Boolean = ThisApplication.SilentOperation
    Dim previousDocument As Document = Nothing
    Try : previousDocument = ThisApplication.ActiveDocument : Catch : End Try
    Dim createdView As Inventor.View = Nothing
    Dim enteredFlatEdit As Boolean = False
    Dim flat As FlatPattern = smDef.FlatPattern
    Try
        ' Export Face As komutu aktif bir Inventor görünümü ve seçili açınım yüzeyi ister.
        ThisApplication.ScreenUpdating = True
        ThisApplication.SilentOperation = False
        If pDoc.Views.Count = 0 Then createdView = pDoc.Views.Add()
        pDoc.Activate()
        flat.Edit()
        enteredFlatEdit = True
        pDoc.SelectSet.Clear()
        pDoc.SelectSet.Select(flat.TopFace)

        Dim commandManager As CommandManager = ThisApplication.CommandManager
        commandManager.ClearPrivateEvents()
        commandManager.PostPrivateEvent(PrivateEventTypeEnum.kFileNameEvent, outputPath)
        Dim exportCommand As ControlDefinition = commandManager.ControlDefinitions.Item("GeomToDXFCommand")
        If Not exportCommand.Enabled Then Throw New System.InvalidOperationException("Inventor Export Face As komutu etkin değil.")
        exportCommand.Execute2(True)

        If Not System.IO.File.Exists(outputPath) OrElse New System.IO.FileInfo(outputPath).Length = 0 Then
            Throw New System.IO.IOException("Inventor Export Face As DXF üretmedi.")
        End If
    Finally
        Try : pDoc.SelectSet.Clear() : Catch : End Try
        If enteredFlatEdit Then
            Try : flat.ExitEdit() : Catch : End Try
        End If
        Try : ThisApplication.CommandManager.ClearPrivateEvents() : Catch : End Try
        If createdView IsNot Nothing Then
            Try
                If pDoc.Views.Count > 1 Then createdView.Close()
            Catch
            End Try
        End If
        If previousDocument IsNot Nothing Then
            Try
                If Not Object.ReferenceEquals(previousDocument, pDoc) Then previousDocument.Activate()
            Catch
            End Try
        End If
        ThisApplication.ScreenUpdating = oldScreenUpdating
        ThisApplication.SilentOperation = oldSilentOperation
    End Try
End Sub

Private Function ValidateClosedDxfContours(filePath As String, ByRef errorText As String, Optional openPoints As List(Of System.Drawing.PointF) = Nothing) As Boolean
    Try
        If openPoints IsNot Nothing Then openPoints.Clear()
        Dim lines As String() = System.IO.File.ReadAllLines(filePath, System.Text.Encoding.Latin1)
        If lines.Length Mod 2 <> 0 Then Throw New System.IO.InvalidDataException("DXF grup çiftleri eksik.")

        Dim endpoints As New Dictionary(Of String, Integer)(StringComparer.Ordinal)
        Dim inEntities As Boolean = False
        Dim awaitingSectionName As Boolean = False
        Dim cutEntityCount As Integer = 0
        Dim polylineActive As Boolean = False
        Dim polylineClosed As Boolean = False
        Dim polylineVertices As New List(Of System.Drawing.PointF)
        Dim i As Integer = 0
        While i + 1 < lines.Length
            Dim code As String = lines(i).Trim()
            Dim value As String = lines(i + 1).Trim()
            If code = "0" AndAlso value = "SECTION" Then
                awaitingSectionName = True
                inEntities = False
                i += 2
                Continue While
            End If
            If awaitingSectionName AndAlso code = "2" Then
                inEntities = (value = "ENTITIES")
                awaitingSectionName = False
                i += 2
                Continue While
            End If
            If code = "0" AndAlso value = "ENDSEC" Then
                inEntities = False
                i += 2
                Continue While
            End If
            If Not inEntities OrElse code <> "0" Then
                i += 2
                Continue While
            End If

            Dim fields As New Dictionary(Of String, String)(StringComparer.Ordinal)
            Dim j As Integer = i + 2
            While j + 1 < lines.Length AndAlso lines(j).Trim() <> "0"
                Dim fieldCode As String = lines(j).Trim()
                If Not fields.ContainsKey(fieldCode) Then fields.Add(fieldCode, lines(j + 1).Trim())
                j += 2
            End While

            Select Case value.ToUpperInvariant()
                Case "LINE"
                    AddDxfEndpoint(endpoints, GetDxfNumber(fields, "10"), GetDxfNumber(fields, "20"))
                    AddDxfEndpoint(endpoints, GetDxfNumber(fields, "11"), GetDxfNumber(fields, "21"))
                    cutEntityCount += 1
                Case "ARC"
                    Dim cx As Double = GetDxfNumber(fields, "10")
                    Dim cy As Double = GetDxfNumber(fields, "20")
                    Dim radius As Double = GetDxfNumber(fields, "40")
                    If radius <= 0 Then Throw New System.IO.InvalidDataException("Geçersiz yay yarıçapı.")
                    Dim a1 As Double = GetDxfNumber(fields, "50") * Math.PI / 180.0
                    Dim a2 As Double = GetDxfNumber(fields, "51") * Math.PI / 180.0
                    AddDxfEndpoint(endpoints, cx + radius * Math.Cos(a1), cy + radius * Math.Sin(a1))
                    AddDxfEndpoint(endpoints, cx + radius * Math.Cos(a2), cy + radius * Math.Sin(a2))
                    cutEntityCount += 1
                Case "CIRCLE"
                    If GetDxfNumber(fields, "40") <= 0 Then Throw New System.IO.InvalidDataException("Geçersiz daire yarıçapı.")
                    cutEntityCount += 1
                Case "SPLINE"
                    Dim splineX As New List(Of Double)
                    Dim splineY As New List(Of Double)
                    For k As Integer = i + 2 To j - 2 Step 2
                        Dim fieldCode As String = lines(k).Trim()
                        If fieldCode = "10" Then
                            splineX.Add(Double.Parse(lines(k + 1).Trim(), System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture))
                        ElseIf fieldCode = "20" Then
                            splineY.Add(Double.Parse(lines(k + 1).Trim(), System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture))
                        End If
                    Next
                    If splineX.Count < 2 OrElse splineX.Count <> splineY.Count Then Throw New System.IO.InvalidDataException("SPLINE kontrol noktaları eksik.")
                    Dim splineFlags As Integer = 0
                    If fields.ContainsKey("70") Then Integer.TryParse(fields("70"), splineFlags)
                    If (splineFlags And 1) = 0 Then
                        AddDxfEndpoint(endpoints, splineX(0), splineY(0))
                        AddDxfEndpoint(endpoints, splineX(splineX.Count - 1), splineY(splineY.Count - 1))
                    End If
                    cutEntityCount += 1
                Case "LWPOLYLINE"
                    Dim vertices As New List(Of System.Drawing.PointF)
                    Dim vx As Double = 0
                    Dim hasX As Boolean = False
                    For k As Integer = i + 2 To j - 2 Step 2
                        Dim fieldCode As String = lines(k).Trim()
                        If fieldCode = "10" Then
                            vx = Double.Parse(lines(k + 1).Trim(), System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture)
                            hasX = True
                        ElseIf fieldCode = "20" AndAlso hasX Then
                            Dim vy As Double = Double.Parse(lines(k + 1).Trim(), System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture)
                            vertices.Add(New System.Drawing.PointF(CSng(vx), CSng(vy)))
                            hasX = False
                        End If
                    Next
                    Dim flags As Integer = 0
                    If fields.ContainsKey("70") Then Integer.TryParse(fields("70"), flags)
                    ValidateDxfPolyline(endpoints, vertices, (flags And 1) <> 0)
                    cutEntityCount += 1
                Case "POLYLINE"
                    If polylineActive Then Throw New System.IO.InvalidDataException("İç içe POLYLINE.")
                    polylineActive = True
                    polylineVertices.Clear()
                    Dim flags As Integer = 0
                    If fields.ContainsKey("70") Then Integer.TryParse(fields("70"), flags)
                    polylineClosed = (flags And 1) <> 0
                Case "VERTEX"
                    If Not polylineActive Then Throw New System.IO.InvalidDataException("POLYLINE dışında VERTEX.")
                    polylineVertices.Add(New System.Drawing.PointF(CSng(GetDxfNumber(fields, "10")), CSng(GetDxfNumber(fields, "20"))))
                Case "SEQEND"
                    If Not polylineActive Then Throw New System.IO.InvalidDataException("POLYLINE dışında SEQEND.")
                    ValidateDxfPolyline(endpoints, polylineVertices, polylineClosed)
                    cutEntityCount += 1
                    polylineActive = False
                Case Else
                    Throw New System.IO.InvalidDataException("Desteklenmeyen DXF nesnesi: " & value)
            End Select
            i = j
        End While

        If polylineActive Then Throw New System.IO.InvalidDataException("POLYLINE SEQEND olmadan bitti.")
        If cutEntityCount = 0 Then Throw New System.IO.InvalidDataException("Kesim geometrisi bulunamadı.")
        Dim hasInvalidEndpoint As Boolean = False
        For Each point As KeyValuePair(Of String, Integer) In endpoints
            If point.Value <> 2 Then
                If Not hasInvalidEndpoint Then errorText = point.Key & " noktasında " & point.Value & " uç var (beklenen: 2)."
                hasInvalidEndpoint = True
                If point.Value = 1 AndAlso openPoints IsNot Nothing Then
                    Dim parts As String() = point.Key.Replace(" mm", "").Split(","c)
                    Dim px As Single = Single.Parse(parts(0).Trim(), System.Globalization.CultureInfo.InvariantCulture)
                    Dim py As Single = Single.Parse(parts(1).Trim(), System.Globalization.CultureInfo.InvariantCulture)
                    openPoints.Add(New System.Drawing.PointF(px, py))
                End If
            End If
        Next
        If hasInvalidEndpoint Then Return False
        errorText = ""
        Return True
    Catch ex As Exception
        errorText = ex.Message
        Return False
    End Try
End Function

Private Sub ValidateDxfPolyline(counts As Dictionary(Of String, Integer), vertices As List(Of System.Drawing.PointF), isClosed As Boolean)
    If vertices.Count < If(isClosed, 3, 2) Then Throw New System.IO.InvalidDataException("POLYLINE köşeleri eksik.")
    If Not isClosed Then
        AddDxfEndpoint(counts, vertices(0).X, vertices(0).Y)
        AddDxfEndpoint(counts, vertices(vertices.Count - 1).X, vertices(vertices.Count - 1).Y)
    End If
End Sub

Private Function GetDxfNumber(fields As Dictionary(Of String, String), code As String) As Double
    If Not fields.ContainsKey(code) Then Throw New System.IO.InvalidDataException("DXF " & code & " koordinatı eksik.")
    Return Double.Parse(fields(code), System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture)
End Function

Private Sub AddDxfEndpoint(counts As Dictionary(Of String, Integer), x As Double, y As Double)
    Dim xText As String = Math.Round(x, 3).ToString("0.000", System.Globalization.CultureInfo.InvariantCulture)
    Dim yText As String = Math.Round(y, 3).ToString("0.000", System.Globalization.CultureInfo.InvariantCulture)
    Dim key As String = xText & ", " & yText & " mm"
    If counts.ContainsKey(key) Then
        counts(key) += 1
    Else
        counts.Add(key, 1)
    End If
End Sub

' ================================================================
'  2) TEKNIK RESIM OLUŞTURMA (.idw)
' ================================================================
Private Sub DoGenerateDrawings(ByRef successCount As Integer, ByRef skippedCount As Integer, dictParts As Dictionary(Of String, Integer), Optional sequenceStart As Integer = 0, Optional totalOverride As Integer = 0)
    Dim outFolder As String = GetSharedOutputFolder()
    gLastOutputFolder = outFolder

    If Not System.IO.File.Exists(gTemplatePath) Then
        AppendLog("[HATA] Şablon dosyası bulunamadı: " & gTemplatePath)
        Return
    End If

    Dim oTG As TransientGeometry = ThisApplication.TransientGeometry
    Dim total As Integer = If(totalOverride > 0, totalOverride, dictParts.Count)
    Dim processed As Integer = sequenceStart
    Dim drawSeq As Integer = sequenceStart

    For Each kvp As KeyValuePair(Of String, Integer) In dictParts
        If gCancelRequested Then
            AppendLog("[BİLGİ] Teknik resim işlemi kullanıcı tarafından durduruldu.")
            Exit For
        End If
        processed += 1
        UpdateProgress(processed, total)
        Dim filePath As String = kvp.Key
        Dim qty As Integer = kvp.Value
        Dim partName As String = System.IO.Path.GetFileNameWithoutExtension(filePath)
        If gLblDxfStatus IsNot Nothing Then gLblDxfStatus.Text = "İşlem Durumu: IDW | " & processed & " / " & total & " | " & partName
        AppendLog("[" & processed & "/" & total & "] " & partName)

        Dim oPartDoc As PartDocument = Nothing
        Dim oDrawDoc As DrawingDocument = Nothing
        Dim openedHere As Boolean = False
        Try
            oPartDoc = TryCast(GetOpenDocumentByPath(filePath), PartDocument)
            If oPartDoc Is Nothing Then
                oPartDoc = CType(ThisApplication.Documents.Open(filePath, False), PartDocument)
                openedHere = True
            End If

            If GetProductionType(oPartDoc) <> PT_SHEETMETAL Then
                skippedCount += 1
                AppendLog("   [UYARI] Üretim tipi Sheet Metal değil, atlandı.")
                Continue For
            End If

            If TypeOf oPartDoc.ComponentDefinition Is SheetMetalComponentDefinition Then
                Dim oSMDef As SheetMetalComponentDefinition = CType(oPartDoc.ComponentDefinition, SheetMetalComponentDefinition)

                Try
                    If Not oSMDef.HasFlatPattern Then
                        oSMDef.Unfold()
                        oSMDef.FlatPattern.ExitEdit()
                    End If
                Catch ex As Exception
                    skippedCount += 1
                    AppendLog("   [HATA] Açınım oluşturulamadı.")
                    Continue For
                End Try

                Dim thkValue As Double = 0
                Try : thkValue = GetThicknessMillimeters(oPartDoc, oSMDef) : Catch : End Try
                If thkValue <= 0 Then
                    skippedCount += 1
                    AppendLog("   [HATA] Geçerli sac kalınlığı okunamadı; teknik resim atlandı.")
                    Continue For
                End If
                Dim thkText As String = Math.Round(thkValue, 2).ToString(System.Globalization.CultureInfo.InvariantCulture)

                Try
                    Dim oProps As PropertySet = oPartDoc.PropertySets.Item("Design Tracking Properties")
                    Dim oAuthorProp As Inventor.Property = Nothing
                    Try : oAuthorProp = oProps.Item("Author") : Catch : End Try
                    If oAuthorProp IsNot Nothing AndAlso String.IsNullOrEmpty(CStr(oAuthorProp.Value)) Then
                        oAuthorProp.Value = gDesignerName
                    End If
                Catch
                End Try

                oDrawDoc = CType(ThisApplication.Documents.Add(DocumentTypeEnum.kDrawingDocumentObject, gTemplatePath, False), DrawingDocument)
                Dim oSheet As Sheet = oDrawDoc.ActiveSheet

                Dim actualTitleBlockTop As Double = TITLEBLOCK_HEIGHT
                Try
                    Dim oTBBox As Box2d = oSheet.TitleBlock.RangeBox
                    If oTBBox IsNot Nothing Then
                        actualTitleBlockTop = Math.Max(oTBBox.MaxPoint.Y, oTBBox.MinPoint.Y)
                    End If
                Catch
                End Try

                Dim availLeft As Double = SHEET_MARGIN
                Dim availBottom As Double = actualTitleBlockTop + SHEET_MARGIN
                Dim availRight As Double = oSheet.Width - SHEET_MARGIN
                Dim availTop As Double = oSheet.Height - SHEET_MARGIN
                Dim availWidth As Double = Math.Max(availRight - availLeft, 5.0)
                Dim availHeight As Double = Math.Max(availTop - availBottom, 5.0)

                Dim oPoint As Point2d = oTG.CreatePoint2d(availLeft + availWidth / 2, availBottom + availHeight / 2)
                Dim oBaseViewOptions As NameValueMap = ThisApplication.TransientObjects.CreateNameValueMap
                oBaseViewOptions.Add("SheetMetalFoldedModel", False)

                Dim oView As DrawingView = oSheet.DrawingViews.AddBaseView(oPartDoc, oPoint, 1.0, _
                    ViewOrientationTypeEnum.kDefaultViewOrientation, DrawingViewStyleEnum.kHiddenLineRemovedDrawingViewStyle, "Default", , oBaseViewOptions)

                Dim scaleX As Double = (availWidth * 0.9) / oView.Width
                Dim scaleY As Double = (availHeight * 0.9) / oView.Height
                oView.Scale = Math.Min(scaleX, scaleY)

                oDrawDoc.Update()
                ThisApplication.UserInterfaceManager.DoEvents()

                drawSeq += 1
                Dim thkFolderText As String = thkText
                Dim targetFolder As String = System.IO.Path.Combine(outFolder, thkFolderText & "MM")
                If Not System.IO.Directory.Exists(targetFolder) Then System.IO.Directory.CreateDirectory(targetFolder)

                Dim idwFileName As String = GetStandardExportBaseName(oPartDoc, filePath, drawSeq, qty) & ".idw"
                Dim idwPath As String = GetAvailableOutputPath(targetFolder, idwFileName)
                Try
                    oDrawDoc.SaveAs(idwPath, False)
                Catch ex As Exception
                    AppendLog("   [HATA] IDW kaydedilemedi: " & idwPath & " - " & ex.Message)
                    skippedCount += 1
                    Continue For
                End Try

                oDrawDoc.Close(True)
                oDrawDoc = Nothing
                successCount += 1
                AppendLog("   [BAŞARILI] Teknik resim üretildi (" & thkFolderText & "mm -> " & thkFolderText & "MM klasörüne)")
            Else
                skippedCount += 1
                AppendLog("   [UYARI] Sac metal değil, atlandı.")
            End If

        Catch ex As Exception
            skippedCount += 1
            AppendLog("   [HATA] " & ex.Message)
        Finally
            If oDrawDoc IsNot Nothing Then Try : oDrawDoc.Close(True) : Catch : End Try
            CloseProcessedPartTab(oPartDoc, openedHere)
        End Try
    Next
End Sub
